import { Request, Response, NextFunction } from 'express';

export const errorHandler = (err: any, req: Request, res: Response, next: NextFunction) => {
    console.error(`[Error Middleware] Ruta: ${req.method} ${req.originalUrl} | Mensaje:`, err);

    // Si es un error de Timeout por parte de SQL Server
    if (err.code === 'ETIMEOUT' || err.number === -2) {
        return res.status(504).json({
            success: false,
            error: 'Gateway Timeout',
            message: 'La base de datos ha tardado demasiado en responder. Por favor, intente nuevamente en unos momentos.'
        });
    }

    // Si es un error de sintaxis o solicitud inválida en SQL Server
    if (err.code === 'EREQUEST' || err.number === 102) {
        return res.status(400).json({
            success: false,
            error: 'Bad Request',
            message: 'Error en la estructura de la consulta a la base de datos.',
            detail: process.env.NODE_ENV === 'development' ? err.message : undefined
        });
    }

    // Error general del servidor
    return res.status(500).json({
        success: false,
        error: 'Internal Server Error',
        message: 'Ocurrió un error interno en el servidor al procesar su solicitud.'
    });
};

export function resolveBucket(path: string): CacheBucket {
  const p = path.toLowerCase();
  if (p.includes("productividad-en-vivo")) return buckets.vivo;
  if (p.includes("balance-gerencial") || p.includes("plazo-fijo")) return buckets.balance;
  if (p.includes("cancelados") || p.includes("avance")) return buckets.gerencial;
  if (p.includes("ranking")) return buckets.ranking;
  if (p.includes("mora") || p.includes("preventiva")) return buckets.mora;
  return buckets.etl;
}

export function invalidateCacheBucket(bucketName?: BucketName) {
  if (!bucketName) {
    Object.values(buckets).forEach((b) => b.store.clear());
    return;
  }
  buckets[bucketName]?.store.clear();
}

async function syncVersionsFromDatabase(): Promise<boolean> {
  try {
    const result = await pool.request().query(`
      SELECT
        [Id],
        CONVERT(VARCHAR(30), [UltimaActualizacion], 126) AS VersionIso,
        FORMAT([UltimaActualizacion], 'dd/MM/yyyy · HH:mm:ss', 'es-PE') AS FechaFormateada,
        ISNULL([Descripcion], '') AS Descripcion
      FROM [DWH_Gestion_Cartera].[dbo].[Log_Actualizacion] WITH (NOLOCK)
      WHERE [Id] IN (1, 2, 3, 4, 5, 6)
    `);

    const rows = result.recordset || [];
    const mapById = new Map<number, any>();
    rows.forEach((r: any) => mapById.set(Number(r.Id), r));

    let anyChanged = false;

    for (const key of ["etl", "gerencial", "ranking", "mora", "vivo", "balance"] as BucketName[]) {
      const bucket = buckets[key];
      const row = mapById.get(bucket.idLog);
      if (!row) continue;

      const latestVer = String(row.VersionIso || "v1");
      bucket.lastFormattedDate = String(row.FechaFormateada || "");
      if (row.Descripcion) {
        bucket.description = String(row.Descripcion);
      }

      if (bucket.version !== latestVer) {
        console.log(
          `🔄 [SmartCache:${key.toUpperCase()} | Id=${bucket.idLog}] Cambio detectado (${bucket.lastFormattedDate}). Refrescando solo sus tablas...`
        );
        await refreshBucketRoutes(bucket, latestVer);
        anyChanged = true;
      }
    }

    return anyChanged;
  } catch (err) {
    console.error("⚠️ [SmartCache] Error leyendo versiones SQL:", err);
    return false;
  }
}

async function refreshBucketRoutes(bucket: CacheBucket, newVersion: string) {
  const routesToWarm = Array.from(bucket.knownRoutes);
  bucket.store.clear();
  bucket.version = newVersion;

  if (routesToWarm.length === 0) return;

  const start = Date.now();
  for (let i = 0; i < routesToWarm.length; i += 4) {
    const batch = routesToWarm.slice(i, i + 4);
    await Promise.allSettled(
      batch.map((route) =>
        fetch(`${lastKnownBaseUrl}${route}`, {
          headers: { "x-internal-prewarm": "true" },
        }).catch(() => {})
      )
    );
  }
  console.log(
    `✅ [SmartCache:${bucket.name.toUpperCase()}] ${routesToWarm.length} rutas actualizadas en RAM en ${((Date.now() - start) / 1000).toFixed(2)}s.`
  );
}

export function startCacheWatcher() {
  const checkLoop = async () => {
    if (isChecking) return;
    isChecking = true;
    try {
      await syncVersionsFromDatabase();
    } finally {
      isChecking = false;
    }
  };

  setTimeout(checkLoop, 1000);
  setInterval(checkLoop, 10000);
}

export async function smartCacheMiddleware(req: Request, res: Response, next: NextFunction) {
  const host = req.headers.host;
  if (host && !process.env.INTERNAL_API_URL) {
    lastKnownBaseUrl = `http://127.0.0.1:${host.split(":")[1] || process.env.PORT || 3000}`;
  }

  // Si es una petición POST, PUT o DELETE, limpiamos inmediatamente el bucket afectado tras completarse
  if (req.method !== "GET") {
    res.on("finish", () => {
      if (res.statusCode >= 200 && res.statusCode < 300) {
        const target = resolveBucket(req.path);
        target.store.clear();
        syncVersionsFromDatabase().catch(() => {});
      }
    });
    return next();
  }

  if (req.path.includes("health")) {
    return next();
  }

  res.setHeader("Cache-Control", "no-store, no-cache, must-revalidate, proxy-revalidate");
  res.setHeader("Pragma", "no-cache");
  res.setHeader("Expires", "0");

  if (req.path.endsWith("/cache-clear")) {
    Object.values(buckets).forEach((b) => b.store.clear());
    return res.json({ ok: true, message: "Los 6 cachés fueron limpiados manualmente" });
  }

  if (req.path.includes("/filtros")) {
    return next();
  }

  const activeBucket = resolveBucket(req.path);
  const cacheKey = req.originalUrl;
  const isInternalPrewarm = req.headers["x-internal-prewarm"] === "true";

  if (!isInternalPrewarm && activeBucket.store.has(cacheKey)) {
    res.setHeader("X-Cache", `HIT-${activeBucket.name.toUpperCase()}`);
    res.setHeader("X-Cache-Bucket", activeBucket.name);
    res.setHeader("X-Data-Version", activeBucket.version);
    return res.json(activeBucket.store.get(cacheKey));
  }

  const originalJson = res.json.bind(res);
  res.json = (body: any) => {
    if (res.statusCode >= 200 && res.statusCode < 300 && !body?.error) {
      activeBucket.store.set(cacheKey, body);
      if (
        activeBucket.knownRoutes.size < 25 &&
        req.originalUrl &&
        (!req.originalUrl.includes("page=") || req.originalUrl.includes("page=1"))
      ) {
        activeBucket.knownRoutes.add(req.originalUrl);
      }
    }
    res.setHeader("X-Cache", isInternalPrewarm ? "PREWARM" : `MISS-${activeBucket.name.toUpperCase()}`);
    res.setHeader("X-Cache-Bucket", activeBucket.name);
    res.setHeader("X-Data-Version", activeBucket.version);
    return originalJson(body);
  };

  next();
}

  }

    const result = await request

      .query(`
        -- =================================================================
        -- 1. ASESORES COMERCIALES
        -- =================================================================
        WITH CTE_Metas_Agencia AS (
            SELECT 
                RIGHT('00' + RTRIM(LTRIM(CAST(IdSAgencia AS VARCHAR(5)))), 2) AS IdSAgencia, 
                ISNULL(MAX(Mora9Meta), 0.10) AS MetaMoraCPP, 
                ISNULL(MAX(Mora31Meta), CASE WHEN @periodo >= '202610' THEN 0.023 ELSE 0.02 END) AS MetaMoraDeficiente
            FROM dm_productividad.dbo.fct_stock_manubl_agencia_month WITH (NOLOCK)
            WHERE Periodo = @periodo 
            GROUP BY RIGHT('00' + RTRIM(LTRIM(CAST(IdSAgencia AS VARCHAR(5)))), 2)
        ),
        CTE_Metas_Asesor AS (
            SELECT 
                RTRIM(LTRIM(IdSAsesor)) AS IdSAsesor, 
                ISNULL(SUM(CarteraInicial), 0) AS CarteraInicial
            FROM dm_productividad.dbo.fct_stock_manubl_asesor_month WITH (NOLOCK)
            WHERE Periodo = @periodo 
            GROUP BY RTRIM(LTRIM(IdSAsesor))
        ),
        CTE_Asesores_Com AS (
            SELECT 
                RTRIM(LTRIM(IdSAsesor)) AS IdSAsesor, 
                MAX(RIGHT('00' + RTRIM(LTRIM(IdSAgencia)), 2)) AS IdSAgencia, 
                MAX(AsesorNombresApellidos) AS Asesor, 
                ISNULL(MAX(ColocacionNumMeta), 30) AS MetaAsesor
            FROM DWH_Gestion_Cartera.dbo.dim_asesor WITH (NOLOCK)
            WHERE Periodo = @periodo AND (Cargo <> 'RECUPERADOR' OR Cargo IS NULL)
            GROUP BY RTRIM(LTRIM(IdSAsesor))
        ),
        CTE_Flow_Com AS (
            SELECT 
                RTRIM(LTRIM(IdSAsesor)) AS IdSAsesor, 
                ISNULL(SUM(ColocacionNumReal), 0) AS NroOper, 
                ISNULL(SUM(ColocacionMontoReal), 0) AS Desembolsos,

            FROM DWH_Gestion_Cartera.dbo.fct_flow WITH (NOLOCK)
            WHERE Periodo = @periodo

            (MS.MetaMoraDeficiente * ISNULL(NULLIF(MS.CarteraInicial, 0), ISNULL(S.Cartera, 0))) AS moraDefMax, 
            ISNULL(S.MoraDeficiente, 0) AS moraDefActual
        FROM CTE_Asesores_Com A 
        LEFT JOIN CTE_Flow_Com F ON A.IdSAsesor = F.IdSAsesor 
        LEFT JOIN CTE_Stock_Com S ON A.IdSAsesor = S.IdSAsesor 
        LEFT JOIN CTE_MetasStock_Com MS ON A.IdSAsesor = MS.IdSAsesor 
        ORDER BY amountAchieved DESC;

        -- =================================================================
        -- 2. RECUPERADORES
        -- =================================================================
        WITH CTE_Metas_Agencia AS (
            SELECT 
                RIGHT('00' + RTRIM(LTRIM(CAST(IdSAgencia AS VARCHAR(5)))), 2) AS IdSAgencia, 
                ISNULL(MAX(Mora9Meta), 0.10) AS MetaMoraCPP, 
                ISNULL(MAX(Mora31Meta), CASE WHEN @periodo >= '202610' THEN 0.023 ELSE 0.02 END) AS MetaMoraDeficiente
            FROM dm_productividad.dbo.fct_stock_manubl_agencia_month WITH (NOLOCK)

        ),
        CTE_Rec_List AS (
            SELECT RTRIM(LTRIM(IdSAsesor)) AS IdSAsesor, MAX(RIGHT('00' + RTRIM(LTRIM(IdSAgencia)), 2)) AS IdSAgencia, MAX(AsesorNombresApellidos) AS Recuperador

            GROUP BY RTRIM(LTRIM(IdSAsesor))

  const asesorParam = (req.query.asesor as string) || "";
  const vista = (req.query.vista as string) || "comercial";

  const page = Math.max(parseInt(req.query.page as string) || 1, 1);
  const limit = Math.max(parseInt(req.query.limit as string) || 100, 1);

    agenciaParam === "Todas" || agenciaParam === "Todas las Agencias"

      vista === "comercial"
        ? `AND DA.Agencia IS NOT NULL AND DA.Agencia NOT IN ('Agencia Desconocida', 'NULL')`
        : `AND (DA.Agencia IS NULL OR DA.Agencia IN ('Agencia Desconocida', 'NULL'))`;

    const querySQL = `
      WITH RiesgoPorPagare AS (
        SELECT
          LTRIM(RTRIM([Pagare])) AS Pagare,
          CAST(ROUND(AVG(CAST(ISNULL([Dias_Atraso], 0) AS FLOAT)), 1) AS DECIMAL(10, 1)) AS Prom_Riesgo,

        FROM [DWH_Gestion_Cartera].[dbo].[fct_riesgo_integral] WITH (NOLOCK)

        SELECT
          ISNULL(DA.Agencia, 'Agencia Desconocida') AS Agencia,
          ISNULL(D_AS.Asesor, A.IdSAsesor) AS Asesor,
          ISNULL(A.[Avance Cartera], 0) AS AvanceCartera,
          COALESCE(

            NULLIF(AD.[Promedio_Dias_Pago], 0),
            RP.Prom_Riesgo,
            0
          ) AS PromedioDiasPago,
          COALESCE(
            NULLIF(A.[Maximo_Dias_Atraso], 0),
            NULLIF(AD.[Maximo_Dias_Atraso], 0),
            RP.Max_Riesgo,
            0
          ) AS MaximoDiasAtraso,

        FROM [DWH_Gestion_Cartera].[dbo].[avance_cartera] A WITH (NOLOCK)
        LEFT JOIN [DWH_Gestion_Cartera].[dbo].[fct_avance_dias_cred] AD WITH (NOLOCK)

      reqEvo.input("periodo", sql.VarChar(6), periodo);
      reqEvo.input("agencia", sql.VarChar(10), codAgenciaFiltro);
      reqEvo.input("todoPreec", sql.Bit, incluirTodoPreec ? 1 : 0);
      reqEvo.input("moneda", sql.VarChar(15), monedaFiltro);

      const exprFactorTC = `CASE WHEN [Moneda] = 'ME' THEN ISNULL([Tipo_Cambio], 3.433) ELSE 1.0 END`;

      const queryEvolutivo =
        granularidad === "Mensual"
          ? `
        WITH ColocMes AS (
          SELECT
            C.[Periodo] AS Etiqueta,
            SUM(C.[Saldo_Colocaciones]) AS Saldo_Colocaciones,
            SUM(C.[Numerador_Saldo_x_TEA]) / NULLIF(SUM(C.[Saldo_Colocaciones]), 0) AS TEA_Activa
          FROM [DWH_Gestion_Cartera].[dbo].[fct_gerencia_colocaciones] C WITH (NOLOCK)
          INNER JOIN (

            FROM [DWH_Gestion_Cartera].[dbo].[fct_gerencia_colocaciones] WITH (NOLOCK)
            GROUP BY [Periodo]
          ) M ON C.[Periodo] = M.[Periodo] AND C.[Fecha_Corte] = M.MaxFecha
          WHERE (@todoPreec = 1 OR C.[Es_Cartera_Comercial] = 1)
            AND (@agencia = '' OR C.${colAgencia} = @agencia)
          GROUP BY C.[Periodo]

            SUM(CASE WHEN LTRIM(RTRIM(S.[Tipo_Aho])) = '03' THEN S.[Saldo_Total] * (${exprFactorTC}) ELSE 0 END) AS Saldo_Plazo_Fijo,
            SUM(S.[Numerador_Saldo_x_TEA] * (${exprFactorTC})) / NULLIF(SUM(S.[Saldo_Total] * (${exprFactorTC})), 0) AS TEA_Pasiva
          FROM [DWH_Gestion_Cartera].[dbo].[fct_gerencia_ahorros_stock] S WITH (NOLOCK)

            ${sqlExcluir09101Stock}
            AND (@agencia = '' OR S.${colAgencia} = @agencia)
            AND (@moneda = 'CONSOLIDADO' OR S.[Moneda] = @moneda)
          GROUP BY S.[Periodo]

        FlujoMes AS (
          SELECT
            [Periodo] AS Etiqueta,
            SUM([Monto_Depositos] * (${exprFactorTC})) AS Depositos,
            SUM([Monto_Retiros] * (${exprFactorTC})) AS Retiros
          FROM [DWH_Gestion_Cartera].[dbo].[fct_gerencia_ahorros_flujo] WITH (NOLOCK)
          WHERE 1 = 1
            ${sqlExcluir09101Flujo}
            AND (@agencia = '' OR ${colAgencia} = @agencia)
            AND (@moneda = 'CONSOLIDADO' OR [Moneda] = @moneda)
          GROUP BY [Periodo]
        ),
        CajaMes AS (
          SELECT
            [Periodo] AS Etiqueta,
            SUM([Saldo_Cierre_MN]) AS Saldo_Caja_020
          FROM [DWH_Gestion_Cartera].[dbo].[fct_gerencia_caja_020] WITH (NOLOCK)
          WHERE [Es_Ultimo_Cierre_Periodo] = 1
            AND (@agencia = '' OR [IdSAgencia_Origen] = @agencia)
          GROUP BY [Periodo]
        ),
        BancosMes AS (
          SELECT
            [Periodo] AS Etiqueta,
            SUM([Saldo_Banco]) AS Saldo_Bancos
          FROM [DWH_Gestion_Cartera].[dbo].[fct_gerencia_bancos_manual] WITH (NOLOCK)
          WHERE (@moneda = 'CONSOLIDADO' OR [Moneda] = @moneda)
          GROUP BY [Periodo]
        ),
        PeriodosBase AS (
          SELECT Etiqueta FROM AhoMes
          UNION SELECT Etiqueta FROM ColocMes
        )
        SELECT TOP 12
          P.Etiqueta AS etiqueta,
          ISNULL(A.Saldo_Ahorros, 0) AS saldoAhorros,
          ISNULL(A.Saldo_Plazo_Fijo, 0) AS saldoPlazoFijo,
          ISNULL(C.Saldo_Colocaciones, 0) AS saldoColocaciones,
          ISNULL(CJ.Saldo_Caja_020, 0) AS saldoCaja020,
          ISNULL(B.Saldo_Bancos, 0) AS saldoBancos,
          ISNULL(F.Depositos, 0) AS depositos,
          ISNULL(F.Retiros, 0) AS retiros,
          ISNULL(C.TEA_Activa, 0) AS teaActiva,
          ISNULL(A.TEA_Pasiva, 0) AS teaPasiva
        FROM PeriodosBase P
        LEFT JOIN ColocMes C ON P.Etiqueta = C.Etiqueta
        LEFT JOIN AhoMes A ON P.Etiqueta = A.Etiqueta
        LEFT JOIN FlujoMes F ON P.Etiqueta = F.Etiqueta
        LEFT JOIN CajaMes CJ ON P.Etiqueta = CJ.Etiqueta
        LEFT JOIN BancosMes B ON P.Etiqueta = B.Etiqueta
        ORDER BY P.Etiqueta ASC;

            SUM([Monto_Depositos] * (${exprFactorTC})) AS Depositos,

          FROM [DWH_Gestion_Cartera].[dbo].[fct_gerencia_ahorros_flujo] WITH (NOLOCK)
          WHERE [Periodo] = @periodo
            ${sqlExcluir09101Flujo}
            AND (@agencia = '' OR ${colAgencia} = @agencia)
            AND (@moneda = 'CONSOLIDADO' OR [Moneda] = @moneda)
          GROUP BY CONVERT(VARCHAR(10), [Fecha_Mov], 23)
        ),
        FlujoColocDia AS (
          SELECT
            CONVERT(VARCHAR(10), [Fecha_Mov], 23) AS Etiqueta,
            SUM(ISNULL([Monto_Desembolsos], 0) - ISNULL([Monto_Repagos], 0)) AS Delta_Colocaciones

          WHERE [Periodo] = @periodo
            AND (@agencia = '' OR [IdSAgencia_Origen] = @agencia)
          GROUP BY CONVERT(VARCHAR(10), [Fecha_Mov], 23)
        ),
        FechasBase AS (
          SELECT Etiqueta FROM FlujoAhoDia
          UNION SELECT Etiqueta FROM FlujoColocDia
          UNION SELECT Etiqueta FROM CajaDia
        )
        SELECT
          F.Etiqueta AS etiqueta,
          ISNULL(FA.Depositos, 0) AS depositos,
          ISNULL(FA.Retiros, 0) AS retiros,
          ISNULL(FA.Delta_Ahorros, 0) AS deltaAhorros,
          ISNULL(FA.Delta_PF, 0) AS deltaPF,
          ISNULL(FC.Delta_Colocaciones, 0) AS deltaColocaciones,
          ISNULL(CJ.Saldo_Caja_020, 0) AS saldoCaja020
        FROM FechasBase F
        LEFT JOIN FlujoAhoDia FA ON F.Etiqueta = FA.Etiqueta
        LEFT JOIN FlujoColocDia FC ON F.Etiqueta = FC.Etiqueta
        LEFT JOIN CajaDia CJ ON F.Etiqueta = CJ.Etiqueta
        ORDER BY F.Etiqueta ASC;
      `;

      const [resAho, resFlujo, resColoc, resCaja, resBancos, resEvo] = await Promise.all([
        reqAho.query(queryAhoStock),
        reqFlujo.query(queryAhoFlujo),
        reqColoc.query(queryColoc),
        reqCaja.query(queryCaja),
        reqBancos.query(queryBancos),

      const ahoRows = resAho.recordset || [];
      const flujoRows = resFlujo.recordset || [];
      const colocRows = resColoc.recordset || [];

      const bancosRows = resBancos.recordset || [];
      const evoRows = resEvo.recordset || [];

      // 9. Consultar Reclasificación de Colocaciones desde el DWH local (CERO conexión a TRANSACMIF)

      try {
        const reqPres = pool.request();
        reqPres.input("periodo", sql.VarChar(6), periodo);
        reqPres.input("fecha", sql.VarChar(10), tieneFechaExacta ? fechaFiltro : "");
        reqPres.input("agencia", sql.VarChar(10), codAgenciaFiltro);
        reqPres.input("asesor", sql.VarChar(50), asesorFiltro === "Todos" ? "" : asesorFiltro);
        reqPres.input("todoPreec", sql.Bit, incluirTodoPreec ? 1 : 0);

        const resPres = await reqPres.query(`
          IF OBJECT_ID('[DWH_Gestion_Cartera].[dbo].[fct_gerencia_colocaciones_tipo]', 'U') IS NOT NULL
          BEGIN
            DECLARE @FechaColocTipo DATE;
            IF @fecha <> '' AND EXISTS (

              FROM [DWH_Gestion_Cartera].[dbo].[fct_gerencia_colocaciones_tipo] WITH (NOLOCK)
              WHERE [Periodo] = @periodo AND [Fecha_Corte] <= CAST(@fecha AS DATE);
            ELSE
              SELECT @FechaColocTipo = MAX([Fecha_Corte])
              FROM [DWH_Gestion_Cartera].[dbo].[fct_gerencia_colocaciones_tipo] WITH (NOLOCK)
              WHERE [Periodo] = @periodo;

            SELECT
              ISNULL(LTRIM(RTRIM([Cod_Tipo_Pres])), '00') AS codTipo,
              ISNULL(LTRIM(RTRIM([Tipo_Prestamo])), 'OTROS') AS tipoPrestamo,
              SUM([Nro_Creditos]) AS nroCreditos,
              SUM([Saldo_Colocaciones]) AS saldoColocaciones,
              SUM([Saldo_Vigente]) AS saldoVigente,
              SUM([Saldo_Refinanciado]) AS saldoRefinanciado,

              SUM([Saldo_Vencido]) AS saldoVencido,
              SUM([Numerador_Saldo_x_TEA]) AS numeradorTea
            FROM [DWH_Gestion_Cartera].[dbo].[fct_gerencia_colocaciones_tipo] WITH (NOLOCK)
            WHERE [Periodo] = @periodo
              AND (@FechaColocTipo IS NULL OR [Fecha_Corte] = @FechaColocTipo)
              AND (@todoPreec = 1 OR [Es_Cartera_Comercial] = 1)
              AND (@agencia = '' OR ${colAgencia} = @agencia)
              AND (@asesor = '' OR [IdSAsesor] = @asesor OR [Nombre_Asesor] = @asesor)
            GROUP BY [Cod_Tipo_Pres], [Tipo_Prestamo]
            ORDER BY saldoColocaciones DESC;
          END
        `);

        porTipoPrestamo = (resPres.recordset || []).map((r: any) => {
          const saldo = Number(r.saldoColocaciones || 0);
          const venc = Number(r.saldoVencido || 0) + Number(r.saldoJudicial || 0);
          return {
            codTipo: r.codTipo,
            tipoPrestamo: r.tipoPrestamo,
            nroCreditos: Number(r.nroCreditos || 0),
            saldoColocaciones: saldo,
            saldoVigente: Number(r.saldoVigente || 0),
            saldoRefinanciado: Number(r.saldoRefinanciado || 0),
            saldoJudicial: Number(r.saldoJudicial || 0),
            saldoVencido: Number(r.saldoVencido || 0),
            pctMora: saldo > 0 ? Number(((venc / saldo) * 100).toFixed(2)) : 0,
            teaPonderada: Number((saldo > 0 ? Number(r.numeradorTea || 0) / saldo : 0).toFixed(2)),
          };
        });
      } catch (errTipo) {
        console.warn("Aviso leyendo fct_gerencia_colocaciones_tipo en DWH:", errTipo);
      }

      let tipoCambioPeriodo = 0;
      for (const r of ahoRows) {
        if (String(r.Moneda).trim() === "ME" && Number(r.Tipo_Cambio) > 0) {
          tipoCambioPeriodo = Number(r.Tipo_Cambio);
          break;
        }
      }
      if (tipoCambioPeriodo === 0 && cajaRows.length > 0) {
        for (const c of cajaRows) {
          if (Number(c.Tipo_Cambio) > 0) {
            tipoCambioPeriodo = Number(c.Tipo_Cambio);
            break;
          }
        }
      }

      const agenciaMap = new Map<string, any>();
      const asesorMap = new Map<string, any>();
      const productoMap = new Map<string, any>();

      const ensureAgencia = (cod: string) => {
        const cleanCod = String(cod || "01").trim().padStart(2, "0");
        if (!agenciaMap.has(cleanCod)) {
          agenciaMap.set(cleanCod, {
            idAgencia: cleanCod,
            agencia: AGENCY_NAMES[cleanCod] || `Agencia ${cleanCod}`,
            saldoAhorros: 0,
            saldoAhorrosUSD: 0,
            saldoAhorrosUSD_Soles: 0,
            numTeaAhorros: 0,
            saldoSoloAhorros: 0,
            saldoSoloAhorrosUSD: 0,
            numTeaSoloAhorros: 0,
            saldoSoloAportes: 0,
            saldoSoloAportesUSD: 0,
            numTeaSoloAportes: 0,
            saldoPlazoFijo: 0,
            saldoPlazoFijoUSD: 0,
            saldoPlazoFijoUSD_Soles: 0,
            numTeaPlazoFijo: 0,
            depositos: 0,
            depositosUSD: 0,
            retiros: 0,
            retirosUSD: 0,
            saldoColocaciones: 0,
            numTeaColocaciones: 0,
            nroCreditos: 0,
            saldoCaja020: 0,
            saldoCaja020USD: 0,
          });
        }
        return agenciaMap.get(cleanCod);
      };

      const ensureAsesor = (idAsesor: string, nombre: string, codAgencia: string) => {
        const key = String(idAsesor || "SISTEMA").trim();
        const cleanCod = String(codAgencia || "01").trim().padStart(2, "0");
        if (!asesorMap.has(key)) {
          asesorMap.set(key, {
            idAsesor: key,
            nombre: String(nombre || key).trim(),
            idAgencia: cleanCod,
            agencia: AGENCY_NAMES[cleanCod] || `Agencia ${cleanCod}`,
            saldoAhorros: 0,
            saldoAhorrosUSD: 0,
            saldoAhorrosUSD_Soles: 0,
            numTeaAhorros: 0,
            saldoSoloAhorros: 0,
            numTeaSoloAhorros: 0,
            saldoSoloAportes: 0,
            numTeaSoloAportes: 0,
            saldoPlazoFijo: 0,
            saldoPlazoFijoUSD: 0,
            saldoPlazoFijoUSD_Soles: 0,
            numTeaPlazoFijo: 0,
            depositos: 0,
            retiros: 0,
            saldoColocaciones: 0,
            numTeaColocaciones: 0,
            nroCreditos: 0,
          });
        }
        return asesorMap.get(key);
      };

      let kpiSaldoAhorrosSoles = 0;
      let kpiSaldoAhorrosUSD = 0;
      let kpiSaldoAhorrosUSD_EnSoles = 0;
      let kpiNumTeaAhorros = 0;

      // Desglose permanente separado:
      // 1) Solo Ahorros (sin 01, sin 09-102 y sin 09-101)
      let kpiSoloAhorrosSoles = 0;
      let kpiSoloAhorrosUSD = 0;
      let kpiSoloAhorrosUSD_EnSoles = 0;
      let kpiNumTeaSoloAhorros = 0;

      // 2) Solo Aportes (01-101, 01-102, 01-103 + 09-102 Ahorro Pres)
      let kpiSoloAportesSoles = 0;
      let kpiSoloAportesUSD = 0;
      let kpiSoloAportesUSD_EnSoles = 0;
      let kpiNumTeaSoloAportes = 0;

      let kpiSaldoPFSoles = 0;
      let kpiSaldoPFUSD = 0;
      let kpiSaldoPFUSD_EnSoles = 0;
      let kpiNumTeaPF = 0;

      ahoRows.forEach((r: any) => {
        const tipoAho = String(r.Tipo_Aho || "").trim();
        const subtipoAho = String(r.Subtipo_Aho || "").trim();

        // Excluir terminantemente 09-101 (29180702 DESEMBOLSO CREDITO)
        if (esSubtipoExcluido(tipoAho, subtipoAho)) return;

        const esME = String(r.Moneda).trim() === "ME";
        const tc = esME ? Number(r.Tipo_Cambio || tipoCambioPeriodo) : 1.0;
        const saldoOrig = Number(r.Saldo_Original || 0);
        const saldoEnSoles = esME ? saldoOrig * tc : saldoOrig;
        const numTeaEnSoles = Number(r.Numerador_TEA_Original || 0) * tc;
        const esPF = tipoAho === "03";
        const esAporte = esSubtipoAporte(tipoAho, subtipoAho);

        const ag = ensureAgencia(r.IdSAgencia);
        const as = ensureAsesor(r.IdSAsesor, r.Nombre_Usuario, r.IdSAgencia);

        if (esPF) {
          kpiSaldoPFSoles += saldoEnSoles;
          kpiNumTeaPF += numTeaEnSoles;
          ag.saldoPlazoFijo += saldoEnSoles;
          ag.numTeaPlazoFijo += numTeaEnSoles;
          as.saldoPlazoFijo += saldoEnSoles;
          as.numTeaPlazoFijo += numTeaEnSoles;

          if (esME) {
            kpiSaldoPFUSD += saldoOrig;
            kpiSaldoPFUSD_EnSoles += saldoEnSoles;
            ag.saldoPlazoFijoUSD += saldoOrig;
            ag.saldoPlazoFijoUSD_Soles += saldoEnSoles;
            as.saldoPlazoFijoUSD += saldoOrig;
            as.saldoPlazoFijoUSD_Soles += saldoEnSoles;
          }
        } else {
          if (esAporte) {
            kpiSoloAportesSoles += saldoEnSoles;
            kpiNumTeaSoloAportes += numTeaEnSoles;
            ag.saldoSoloAportes += saldoEnSoles;
            ag.numTeaSoloAportes += numTeaEnSoles;
            as.saldoSoloAportes += saldoEnSoles;
            as.numTeaSoloAportes += numTeaEnSoles;
            if (esME) {
              kpiSoloAportesUSD += saldoOrig;
              kpiSoloAportesUSD_EnSoles += saldoEnSoles;
              ag.saldoSoloAportesUSD += saldoOrig;
            }
          } else {
            kpiSoloAhorrosSoles += saldoEnSoles;
            kpiNumTeaSoloAhorros += numTeaEnSoles;
            ag.saldoSoloAhorros += saldoEnSoles;
            ag.numTeaSoloAhorros += numTeaEnSoles;
            as.saldoSoloAhorros += saldoEnSoles;
            as.numTeaSoloAhorros += numTeaEnSoles;
            if (esME) {
              kpiSoloAhorrosUSD += saldoOrig;
              kpiSoloAhorrosUSD_EnSoles += saldoEnSoles;
              ag.saldoSoloAhorrosUSD += saldoOrig;
            }
          }

          const incluirEnActivo =
            tipoCaptacion === "COMBINADO" ||
            (tipoCaptacion === "APORTES" && esAporte) ||
            (tipoCaptacion === "AHORROS" && !esAporte);

          if (incluirEnActivo) {
            kpiSaldoAhorrosSoles += saldoEnSoles;
            kpiNumTeaAhorros += numTeaEnSoles;
            ag.saldoAhorros += saldoEnSoles;
            ag.numTeaAhorros += numTeaEnSoles;
            as.saldoAhorros += saldoEnSoles;
            as.numTeaAhorros += numTeaEnSoles;

            if (esME) {
              kpiSaldoAhorrosUSD += saldoOrig;
              kpiSaldoAhorrosUSD_EnSoles += saldoEnSoles;
              ag.saldoAhorrosUSD += saldoOrig;
              ag.saldoAhorrosUSD_Soles += saldoEnSoles;
              as.saldoAhorrosUSD += saldoOrig;
              as.saldoAhorrosUSD_Soles += saldoEnSoles;
            }
          }
        }

        const mostrarProducto =
          esPF ||
          tipoCaptacion === "COMBINADO" ||
          (tipoCaptacion === "APORTES" && esAporte) ||
          (tipoCaptacion === "AHORROS" && !esAporte);

        if (mostrarProducto) {
          const prodKey = `${tipoAho}-${subtipoAho}`;
          if (!productoMap.has(prodKey)) {
            productoMap.set(prodKey, {
              codigo: prodKey,
              tipoAho,
              subtipoAho,
              categoria: esPF ? "PLAZO FIJO" : esAporte ? "APORTES (31110101)" : "AHORROS",
              producto: r.Nombre_Producto,
              moneda: r.Moneda,
              tipoCambio: Number(tc.toFixed(4)),
              nroCuentas: 0,
              saldoOriginal: 0,
              saldoTotal: 0,
              numTea: 0,
              depositosOriginal: 0,
              depositos: 0,
              retirosOriginal: 0,
              retiros: 0,
            });
          }
          const prod = productoMap.get(prodKey);
          prod.nroCuentas += Number(r.Nro_Cuentas || 0);
          prod.saldoOriginal += saldoOrig;
          prod.saldoTotal += saldoEnSoles;
          prod.numTea += numTeaEnSoles;
        }
      });

      let kpiDepositosSoles = 0;
      let kpiDepositosUSD = 0;
      let kpiRetirosSoles = 0;
      let kpiRetirosUSD = 0;

      flujoRows.forEach((r: any) => {
        const tipoAho = String(r.Tipo_Aho || "").trim();
        const subtipoAho = String(r.Subtipo_Aho || "").trim();

        if (esSubtipoExcluido(tipoAho, subtipoAho)) return;

        const esPF = tipoAho === "03";
        const esAporte = esSubtipoAporte(tipoAho, subtipoAho);

        const incluirFlujo =
          esPF ||
          tipoCaptacion === "COMBINADO" ||
          (tipoCaptacion === "APORTES" && esAporte) ||
          (tipoCaptacion === "AHORROS" && !esAporte);

        if (!incluirFlujo) return;

        const esME = String(r.Moneda).trim() === "ME";
        const tc = esME ? Number(r.Tipo_Cambio || tipoCambioPeriodo) : 1.0;
        const depOrig = Number(r.Monto_Depositos || 0);
        const retOrig = Number(r.Monto_Retiros || 0);
        const depSoles = esME ? depOrig * tc : depOrig;
        const retSoles = esME ? retOrig * tc : retOrig;

        kpiDepositosSoles += depSoles;
        kpiRetirosSoles += retSoles;
        if (esME) {
          kpiDepositosUSD += depOrig;
          kpiRetirosUSD += retOrig;
        }

        const ag = ensureAgencia(r.IdSAgencia);
        ag.depositos += depSoles;
        ag.retiros += retSoles;
        if (esME) {
          ag.depositosUSD += depOrig;
          ag.retirosUSD += retOrig;
        }

        const as = ensureAsesor(r.IdSAsesor, r.Nombre_Usuario, r.IdSAgencia);
        as.depositos += depSoles;
        as.retiros += retSoles;

        const prodKey = `${tipoAho}-${subtipoAho}`;
        if (productoMap.has(prodKey)) {
          const prod = productoMap.get(prodKey);
          prod.depositosOriginal += depOrig;
          prod.depositos += depSoles;
          prod.retirosOriginal += retOrig;
          prod.retiros += retSoles;
        }
      });

      let kpiSaldoColoc = 0;
      let kpiNumTeaColoc = 0;
      let kpiNroCreditos = 0;

      colocRows.forEach((r: any) => {
        const saldo = Number(r.Saldo_Colocaciones || 0);
        const numTea = Number(r.Numerador_TEA || 0);
        const creds = Number(r.Nro_Creditos || 0);

        kpiSaldoColoc += saldo;
        kpiNumTeaColoc += numTea;
        kpiNroCreditos += creds;

        const ag = ensureAgencia(r.IdSAgencia);
        ag.saldoColocaciones += saldo;
        ag.numTeaColocaciones += numTea;
        ag.nroCreditos += creds;

        const as = ensureAsesor(r.IdSAsesor, r.Nombre_Asesor, r.IdSAgencia);
        as.saldoColocaciones += saldo;
        as.numTeaColocaciones += numTea;
        as.nroCreditos += creds;
      });

      let kpiSaldoCaja020 = 0;
      let kpiSaldoCaja020USD = 0;
      let kpiSaldoCaja020USD_EnSoles = 0;

      cajaRows.forEach((r: any) => {
        const cierreMN = Number(r.Saldo_Cierre_MN || 0);
        const cierreME = Number(r.Saldo_Cierre_ME || 0);
        const cierreME_Soles = cierreME * tipoCambioPeriodo;

        const ag = ensureAgencia(r.IdSAgencia);
        ag.saldoCaja020USD += cierreME;

        if (monedaFiltro === "ME") {
          kpiSaldoCaja020 += cierreME_Soles;
          kpiSaldoCaja020USD += cierreME;
          kpiSaldoCaja020USD_EnSoles += cierreME_Soles;
          ag.saldoCaja020 += cierreME_Soles;
        } else if (monedaFiltro === "CONSOLIDADO") {
          kpiSaldoCaja020 += cierreMN + cierreME_Soles;
          kpiSaldoCaja020USD += cierreME;
          kpiSaldoCaja020USD_EnSoles += cierreME_Soles;
          ag.saldoCaja020 += cierreMN + cierreME_Soles;
        } else {
          kpiSaldoCaja020 += cierreMN;
          ag.saldoCaja020 += cierreMN;
        }
      });

      let kpiSaldoBancos = 0;
      let kpiSaldoBancosUSD = 0;
      bancosRows.forEach((b: any) => {
        const esME = String(b.Moneda).trim() === "ME";
        const monto = Number(b.Saldo_Banco || 0);
        if (esME) {
          kpiSaldoBancosUSD += monto;
          kpiSaldoBancos += monto * tipoCambioPeriodo;
        } else {
          kpiSaldoBancos += monto;
        }
      });

      const teaPasivaAhorros =
        kpiSaldoAhorrosSoles > 0 ? kpiNumTeaAhorros / kpiSaldoAhorrosSoles : 0;
      const teaSoloAhorros =
        kpiSoloAhorrosSoles > 0 ? kpiNumTeaSoloAhorros / kpiSoloAhorrosSoles : 0;
      const teaSoloAportes =
        kpiSoloAportesSoles > 0 ? kpiNumTeaSoloAportes / kpiSoloAportesSoles : 0;
      const teaPasivaPlazoFijo =
        kpiSaldoPFSoles > 0 ? kpiNumTeaPF / kpiSaldoPFSoles : 0;
      const saldoCaptacionesTotal = kpiSaldoAhorrosSoles + kpiSaldoPFSoles;
      const teaPasivaGlobal =
        saldoCaptacionesTotal > 0
          ? (kpiNumTeaAhorros + kpiNumTeaPF) / saldoCaptacionesTotal
          : 0;
      const teaActivaPonderada =
        kpiSaldoColoc > 0 ? kpiNumTeaColoc / kpiSaldoColoc : 0;
      const spreadFinanciero = teaActivaPonderada - teaPasivaGlobal;

      const porAgencia = Array.from(agenciaMap.values())
        .map((a) => {
          const totalCap = a.saldoAhorros + a.saldoPlazoFijo;
          const teaAho = a.saldoAhorros > 0 ? a.numTeaAhorros / a.saldoAhorros : 0;
          const teaSoloAho =
            a.saldoSoloAhorros > 0 ? a.numTeaSoloAhorros / a.saldoSoloAhorros : 0;
          const teaSoloApo =
            a.saldoSoloAportes > 0 ? a.numTeaSoloAportes / a.saldoSoloAportes : 0;
          const teaPF =
            a.saldoPlazoFijo > 0 ? a.numTeaPlazoFijo / a.saldoPlazoFijo : 0;
          const teaPasiva =
            totalCap > 0 ? (a.numTeaAhorros + a.numTeaPlazoFijo) / totalCap : 0;
          const teaActiva =
            a.saldoColocaciones > 0
              ? a.numTeaColocaciones / a.saldoColocaciones
              : 0;
          return {
            ...a,
            totalCaptaciones: totalCap,
            teaAhorros: Number(teaAho.toFixed(2)),
            teaSoloAhorros: Number(teaSoloAho.toFixed(2)),
            teaSoloAportes: Number(teaSoloApo.toFixed(2)),
            teaPlazoFijo: Number(teaPF.toFixed(2)),
            teaPasivaTotal: Number(teaPasiva.toFixed(2)),
            teaActiva: Number(teaActiva.toFixed(2)),
            spread: Number((teaActiva - teaPasiva).toFixed(2)),
          };
        })
        .sort((a, b) => a.idAgencia.localeCompare(b.idAgencia));

      const porAsesor = Array.from(asesorMap.values())
        .map((as) => {
          const totalCap = as.saldoAhorros + as.saldoPlazoFijo;
          const teaAho =
            as.saldoAhorros > 0 ? as.numTeaAhorros / as.saldoAhorros : 0;
          const teaSoloAho =
            as.saldoSoloAhorros > 0 ? as.numTeaSoloAhorros / as.saldoSoloAhorros : 0;
          const teaSoloApo =
            as.saldoSoloAportes > 0 ? as.numTeaSoloAportes / as.saldoSoloAportes : 0;
          const teaPF =
            as.saldoPlazoFijo > 0 ? as.numTeaPlazoFijo / as.saldoPlazoFijo : 0;
          const teaPasiva =
            totalCap > 0
              ? (as.numTeaAhorros + as.numTeaPlazoFijo) / totalCap
              : 0;
          const teaActiva =
            as.saldoColocaciones > 0
              ? as.numTeaColocaciones / as.saldoColocaciones
              : 0;
          return {
            ...as,
            totalCaptaciones: totalCap,
            teaAhorros: Number(teaAho.toFixed(2)),
            teaSoloAhorros: Number(teaSoloAho.toFixed(2)),
            teaSoloAportes: Number(teaSoloApo.toFixed(2)),
            teaPlazoFijo: Number(teaPF.toFixed(2)),
            teaPasivaTotal: Number(teaPasiva.toFixed(2)),
            teaActiva: Number(teaActiva.toFixed(2)),
          };
        })
        .sort(
          (a, b) =>
            b.saldoColocaciones +
            b.totalCaptaciones -
            (a.saldoColocaciones + a.totalCaptaciones)
        );

      const porProductoAhorro = Array.from(productoMap.values())
        .map((p) => ({
          ...p,
          teaPonderada: Number(
            (p.saldoTotal > 0 ? p.numTea / p.saldoTotal : 0).toFixed(2)
          ),
          flujoNeto: p.depositos - p.retiros,
        }))
        .sort((a, b) => b.saldoTotal - a.saldoTotal);

      let serieEvolutivaFinal: any[] = [];
      if (granularidad === "Diario" && evoRows.length > 0) {
        let runningAho = kpiSaldoAhorrosSoles;
        let runningPF = kpiSaldoPFSoles;
        let runningColoc = kpiSaldoColoc;
        let lastValidCaja = kpiSaldoCaja020;

        const reversed = [...evoRows].reverse();
        const reconstructedReversed = reversed.map((dia: any) => {
          const cajaDia = Number(dia.saldoCaja020 || 0);
          if (cajaDia > 0) {
            lastValidCaja = cajaDia;
          }
          const item = {
            etiqueta: dia.etiqueta,
            saldoAhorros: Math.max(0, Number(runningAho.toFixed(2))),
            saldoPlazoFijo: Math.max(0, Number(runningPF.toFixed(2))),
            saldoColocaciones: Math.max(0, Number(runningColoc.toFixed(2))),
            saldoCaja020: cajaDia > 0 ? cajaDia : lastValidCaja,
            saldoBancos: kpiSaldoBancos,
            depositos: Number(dia.depositos || 0),
            retiros: Number(dia.retiros || 0),
            teaActiva: Number(teaActivaPonderada.toFixed(2)),
            teaPasiva: Number(teaPasivaGlobal.toFixed(2)),
            spread: Number(spreadFinanciero.toFixed(2)),
          };
          runningAho -= Number(dia.deltaAhorros || 0);
          runningPF -= Number(dia.deltaPF || 0);
          runningColoc -= Number(dia.deltaColocaciones || 0);
          return item;
        });
        serieEvolutivaFinal = reconstructedReversed.reverse();
      } else {
        serieEvolutivaFinal = evoRows.map((e: any) => ({
          etiqueta: e.etiqueta,
          saldoAhorros: Number(e.saldoAhorros || 0),
          saldoPlazoFijo: Number(e.saldoPlazoFijo || 0),
          saldoColocaciones: Number(e.saldoColocaciones || 0),
          saldoCaja020: Number(e.saldoCaja020 || 0),
          saldoBancos: Number(e.saldoBancos || 0),
          depositos: Number(e.depositos || 0),
          retiros: Number(e.retiros || 0),
          teaActiva: Number(Number(e.teaActiva || 0).toFixed(2)),
          teaPasiva: Number(Number(e.teaPasiva || 0).toFixed(2)),
          spread: Number(
            (Number(e.teaActiva || 0) - Number(e.teaPasiva || 0)).toFixed(2)
          ),
        }));
      }

      res.json({
        periodoActual: periodo,
        monedaActual: monedaFiltro,
        tipoCambio: Number(tipoCambioPeriodo.toFixed(4)),
        filtros: {
          periodos: listaPeriodos.length > 0 ? listaPeriodos : [periodoDefault],
          fechasDisponibles,
          asesores: porAsesor.map((a) => ({
            id: a.idAsesor,
            nombre: a.nombre,
            agencia: a.agencia,
          })),
        },
        kpis: {
          tipoCambio: Number(tipoCambioPeriodo.toFixed(4)),
          saldoAhorros: kpiSaldoAhorrosSoles,
          saldoAhorrosUSD: kpiSaldoAhorrosUSD,
          saldoAhorrosUSD_EnSoles: kpiSaldoAhorrosUSD_EnSoles,
          teaPasivaAhorros: Number(teaPasivaAhorros.toFixed(2)),
          saldoSoloAhorros: kpiSoloAhorrosSoles,
          saldoSoloAhorrosUSD: kpiSoloAhorrosUSD,
          saldoSoloAhorrosUSD_EnSoles: kpiSoloAhorrosUSD_EnSoles,
          teaSoloAhorros: Number(teaSoloAhorros.toFixed(2)),
          saldoSoloAportes: kpiSoloAportesSoles,
          saldoSoloAportesUSD: kpiSoloAportesUSD,
          saldoSoloAportesUSD_EnSoles: kpiSoloAportesUSD_EnSoles,
          teaSoloAportes: Number(teaSoloAportes.toFixed(2)),
          saldoPlazoFijo: kpiSaldoPFSoles,
          saldoPlazoFijoUSD: kpiSaldoPFUSD,
          saldoPlazoFijoUSD_EnSoles: kpiSaldoPFUSD_EnSoles,
          teaPasivaPlazoFijo: Number(teaPasivaPlazoFijo.toFixed(2)),
          saldoCaptacionesTotal,
          saldoCaptacionesUSD: kpiSaldoAhorrosUSD + kpiSaldoPFUSD,
          teaPasivaGlobal: Number(teaPasivaGlobal.toFixed(2)),
          depositosFecha: kpiDepositosSoles,
          depositosFechaUSD: kpiDepositosUSD,
          retirosFecha: kpiRetirosSoles,
          retirosFechaUSD: kpiRetirosUSD,
          flujoNetoAhorros: kpiDepositosSoles - kpiRetirosSoles,
          flujoNetoAhorrosUSD: kpiDepositosUSD - kpiRetirosUSD,
          saldoColocaciones: kpiSaldoColoc,
          teaActivaPonderada: Number(teaActivaPonderada.toFixed(2)),
          spreadFinanciero: Number(spreadFinanciero.toFixed(2)),

          FROM [dm_productividad].[dbo].[dim_calendario] WITH (NOLOCK)
          WHERE [Fecha] <= CAST(DATEADD(hour, -5, GETUTCDATE()) AS DATE)
          ORDER BY [Fecha] DESC

import { Router, Request, Response } from "express";
import sql from "mssql";
import { pool } from "../lib/db";

const router = Router();

router.get(["/gestion-preventiva", "/api/gestion-preventiva"], async (req: Request, res: Response) => {
  const agenciaParam = ((req.query.agencia as string) || "").trim();
  const analistaParam = ((req.query.analista as string) || (req.query.asesor as string) || "").trim();

  const agencia = agenciaParam === "Todas" || agenciaParam === "Todas las Agencias" ? "" : agenciaParam;
  const analista = analistaParam === "Todos" || analistaParam === "Todos los Analistas" ? "" : analistaParam;

  try {
    const result = await pool
      .request()
      .input("agencia", sql.VarChar(60), agencia)
      .input("analista", sql.VarChar(150), analista)
      .query(`
        DECLARE @UltimaFecha DATE = (
          SELECT MAX([Fecha_Foto]) FROM [DWH_Gestion_Cartera].[dbo].[fct_riesgo_integral] WITH (NOLOCK)
        );

        WITH BasePreventivaDWH AS (
          SELECT
            CASE R.[IdSAgencia]
              WHEN '01' THEN 'Wanchaq' WHEN '02' THEN 'San Jerónimo' WHEN '03' THEN 'Quillabamba'
              WHEN '04' THEN 'Sicuani' WHEN '05' THEN 'Molino' WHEN '06' THEN 'Juliaca'
              WHEN '07' THEN 'Lima Los Olivos' WHEN '08' THEN 'Tica Tica' WHEN '09' THEN 'Magisterio'
              WHEN '10' THEN 'Lima SJL' WHEN '11' THEN 'Chiclayo' WHEN '12' THEN 'Arequipa'
              WHEN '13' THEN 'Pucallpa' ELSE 'Otra Agencia'
            END AS [agencia],
            RTRIM(LTRIM(R.[Socio])) AS [socio],
            ISNULL(NULLIF(LTRIM(RTRIM(R.[Telefonos])), ''), 'SIN TELEFONO') AS [telefono],
            ISNULL(R.[Producto], '') AS [producto],
            ISNULL(R.[Saldo_Capital], 0) AS [saldo_numerico],
            FORMAT(ISNULL(R.[Saldo_Capital], 0), 'N2', 'es-PE') AS [saldo_formateado],
            ISNULL(D_AS.NombreAsesor, RTRIM(LTRIM(R.[IdSAsesor]))) AS [analista],
            CONVERT(VARCHAR(10), R.[Fecha_Foto], 103) AS [fecha_vencimiento],
            ISNULL(R.[Dias_Atraso], 0) AS [dias_para_vencimiento]
          FROM [DWH_Gestion_Cartera].[dbo].[fct_riesgo_integral] R WITH (NOLOCK)
          OUTER APPLY (
            SELECT TOP 1 ISNULL(DA.[AsesorNombresApellidos], DA.[Asesor]) AS NombreAsesor
            FROM [DWH_Gestion_Cartera].[dbo].[dim_asesor] DA WITH (NOLOCK)
            WHERE RTRIM(LTRIM(DA.[IdSAsesor])) = RTRIM(LTRIM(R.[IdSAsesor]))
            ORDER BY DA.[Periodo] DESC
          ) D_AS
          WHERE R.[Fecha_Foto] = @UltimaFecha
            AND R.[Probabilidad_IA] >= 30
        )
        SELECT * FROM BasePreventivaDWH
        WHERE (@agencia = '' OR [agencia] = @agencia)
          AND (@analista = '' OR [analista] LIKE '%' + @analista + '%')
        ORDER BY [saldo_numerico] DESC;
      `);

    res.json(result.recordset || []);
  } catch (error) {
    console.error("Error en /gestion-preventiva:", error);
    res.status(500).json({ error: "Error obteniendo datos de Gestión Preventiva desde DWH" });
  }
});

import { Router, Request, Response } from "express";

import { pool } from "../lib/db";

const router = Router();

router.get(["/productividad-en-vivo", "/api/productividad-en-vivo"], async (req: Request, res: Response) => {
  const periodo = ((req.query.periodo as string) || "").trim();

  try {
    const result = await pool.request()
      .input("periodo", sql.VarChar(6), periodo)
      .query(`
        DECLARE @Hoy DATE = CAST(DATEADD(hour, -5, GETUTCDATE()) AS DATE);
        DECLARE @PeriodoDefault VARCHAR(6) = CASE WHEN @periodo <> '' THEN @periodo ELSE CONVERT(VARCHAR(6), @Hoy, 112) END;

        WITH Agencias AS (
          SELECT * FROM (VALUES
            ('01', 'Wanchaq'), ('02', 'San Jerónimo'), ('03', 'Quillabamba'),
            ('04', 'Sicuani'), ('05', 'Molino'), ('06', 'Juliaca'),
            ('07', 'Lima Los Olivos'), ('08', 'Tica Tica'), ('09', 'Magisterio'),
            ('10', 'Lima SJL'), ('11', 'Chiclayo'), ('12', 'Arequipa'),
            ('13', 'Pucallpa')
          ) AS t(IdSAgencia, NombreAgencia)
        ),
        MetasHoy AS (
          SELECT
            RIGHT('00' + CAST(IdSAgencia AS VARCHAR(2)), 2) AS IdSAgencia,
            ISNULL(ColocacionNumProy, 0) AS proyOpDia,
            ISNULL(ColocacionMontoProy, 0) AS proyMontoDia
          FROM [dm_productividad].[dbo].[FctDiario_MetaProy] WITH (NOLOCK)
          WHERE Fecha = @Hoy
        ),
        ControlHoy AS (
          SELECT
            IdSAgencia,
            Proy_Op_Admin,
            Proy_Monto_Admin,
            Sugerencia_Op_Gerencia,
            Sugerencia_Monto_Gerencia
          FROM [DWH_Gestion_Cartera].[dbo].[fct_metas_diarias_control] WITH (NOLOCK)
          WHERE Fecha = @Hoy
        ),
        VivoHoy AS (
          SELECT
            RTRIM(LTRIM(IdSAgencia)) AS IdSAgencia,
            CONVERT(VARCHAR(10), [Fecha], 120) AS Fecha,
            [Periodo],
            SUM([ColocacionNumReal]) AS colocacionNumReal,
            SUM([ColocacionMontoReal]) AS colocacionMontoReal
          FROM [DWH_Gestion_Cartera].[dbo].[fct_productividad_en_vivo_log] WITH (NOLOCK)
          WHERE [Fecha] = @Hoy
          GROUP BY RTRIM(LTRIM(IdSAgencia)), [Fecha], [Periodo]
        )
        SELECT
          ISNULL(V.Fecha, CONVERT(VARCHAR(10), @Hoy, 120)) AS fecha,
          ISNULL(V.Periodo, @PeriodoDefault) AS periodo,
          A.IdSAgencia AS idAgencia,
          A.NombreAgencia AS agencia,
          COALESCE(C.Proy_Op_Admin, M.proyOpDia, C.Sugerencia_Op_Gerencia, 0) AS proyeccionColocacion,
          COALESCE(C.Proy_Monto_Admin, M.proyMontoDia, C.Sugerencia_Monto_Gerencia, 0) AS proyeccionMonto,
          ISNULL(V.colocacionNumReal, 0) AS colocacionNumReal,
          ISNULL(V.colocacionMontoReal, 0) AS colocacionMontoReal
        FROM Agencias A
        LEFT JOIN MetasHoy M ON A.IdSAgencia = M.IdSAgencia
        LEFT JOIN ControlHoy C ON A.IdSAgencia = C.IdSAgencia
        LEFT JOIN VivoHoy V ON A.IdSAgencia = V.IdSAgencia
        ORDER BY A.IdSAgencia ASC;
      `);

    const logRes = await pool.request().query(`
      SELECT TOP (1) 
        FORMAT([UltimaActualizacion], 'dd/MM/yyyy · HH:mm:ss', 'es-PE') AS UltimaActualizacion,
        ISNULL([Descripcion], 'Productividad en Vivo (Transacmif)') AS Descripcion
      FROM [DWH_Gestion_Cartera].[dbo].[Log_Actualizacion] WITH (NOLOCK)
      WHERE [Id] = 5
    `);

    const rowLog = logRes.recordset?.[0];
    const ultimaAct = rowLog?.UltimaActualizacion || "Sincronizando...";
    const desc = rowLog?.Descripcion || "Productividad en Vivo (Transacmif)";

    res.json({
      rows: result.recordset || [],
      ultimaActualizacionTexto: `${desc}: ${ultimaAct}`,
    });
  } catch (error) {
    console.error("Error consultando productividad en vivo:", error);
    res.status(500).json({ error: "Error al obtener productividad en vivo" });
  }
});

import { Router, Request, Response } from "express";
import { pool } from "../lib/db";

const router = Router();

router.get(["/ranking", "/api/ranking"], async (_req: Request, res: Response) => {
  try {
    const request = pool.request();

    // Deduplica por IdSAsesor tomando únicamente el registro más reciente y mayor puntaje
    const querySQL = `
      WITH UltimoPorAsesor AS (
        SELECT
          CONVERT(VARCHAR(10), CAST([Fecha_Calculo] AS DATE), 23) + 'T12:00:00' AS [Fecha_Calculo],
          RTRIM(LTRIM([IdSAsesor])) AS [IdSAsesor],
          RTRIM(LTRIM([Asesor])) AS [Asesor],
          RTRIM(LTRIM([Agencia])) AS [Agencia],
          ISNULL([Operaciones], 0) AS [Operaciones],
          ISNULL([Crecimiento_Neto_150], 0) AS [Crecimiento_Neto_150],
          ISNULL([Seguros_Vendidos], 0) AS [Seguros_Vendidos],
          ISNULL([Puntos_Op], 0) AS [Puntos_Op],
          ISNULL([Puntos_Crec], 0) AS [Puntos_Crec],
          ISNULL([Puntos_Seg], 0) AS [Puntos_Seg],
          ISNULL([Puntaje_Total], 0) AS [Puntaje_Total],
          [Puesto_Actual] AS [Puesto_Original],
          [Puesto_Anterior],
          [Comentario_Creativo],
          ROW_NUMBER() OVER (
            PARTITION BY RTRIM(LTRIM(ISNULL(NULLIF([IdSAsesor], ''), [Asesor])))
            ORDER BY CAST([Fecha_Calculo] AS DATETIME) DESC, [Puntaje_Total] DESC, [Operaciones] DESC
          ) AS rn
        FROM [DWH_Gestion_Cartera].[dbo].[ranking_asesor_10] WITH (NOLOCK)
      )
      SELECT
        [Fecha_Calculo],
        [IdSAsesor],
        [Asesor],
        [Agencia],
        [Operaciones],
        [Crecimiento_Neto_150],
        [Seguros_Vendidos],
        [Puntos_Op],
        [Puntos_Crec],
        [Puntos_Seg],
        [Puntaje_Total],
        CAST(
          RANK() OVER (ORDER BY [Puntaje_Total] DESC, [Operaciones] DESC, [Crecimiento_Neto_150] DESC)
          AS INT
        ) AS [Puesto_Actual],
        [Puesto_Anterior],
        [Comentario_Creativo]
      FROM UltimoPorAsesor
      WHERE rn = 1
      ORDER BY [Puesto_Actual] ASC, [Operaciones] DESC;
    `;

    const result = await request.query(querySQL);
    res.json(result.recordset);
  } catch (error) {
    console.error("Error obteniendo Ranking de Excelencia:", error);
    res.status(500).json({ error: "Error al consultar la base de datos de ranking" });
  }
});

import { Router, Request, Response } from "express";
import sql from "mssql";
import { pool } from "../lib/db";

const router = Router();

router.get(["/supervision/:periodo", "/api/supervision/:periodo"], async (req: Request, res: Response) => {
  const { periodo } = req.params;

  try {
    const request = pool.request();

    const result = await request
      .input("periodo", sql.VarChar(6), periodo)
      .query(`
        -- =================================================================
        -- 1. AGENCIA COMPLETA (Comerciales + Recuperadores)
        -- =================================================================
        WITH CTE_CarteraInic_All AS (
            SELECT 
                DA.IdSAgencia, 
                SUM(A.CarteraInicial) AS CarteraInicial
            FROM dm_productividad.dbo.fct_stock_manubl_asesor_month A WITH (NOLOCK)
            INNER JOIN (
                SELECT RTRIM(LTRIM(IdSAsesor)) AS IdSAsesor, MAX(RIGHT('00' + RTRIM(LTRIM(IdSAgencia)), 2)) AS IdSAgencia
                FROM DWH_Gestion_Cartera.dbo.dim_asesor WITH (NOLOCK)
                WHERE Periodo = @periodo
                GROUP BY RTRIM(LTRIM(IdSAsesor))
            ) DA ON RTRIM(LTRIM(A.IdSAsesor)) = DA.IdSAsesor
            WHERE A.Periodo = @periodo
            GROUP BY DA.IdSAgencia
        ),
        CTE_Flow_All AS (
            SELECT RIGHT('00' + RTRIM(LTRIM(IdSAgencia)), 2) AS IdSAgencia, ISNULL(SUM(ColocacionNumReal), 0) AS NroOper, ISNULL(SUM(ColocacionMontoReal), 0) AS Desembolsos, ISNULL(SUM(RepagoReal), 0) AS Repagos 
            FROM DWH_Gestion_Cartera.dbo.fct_flow WITH (NOLOCK) WHERE Periodo = @periodo GROUP BY RIGHT('00' + RTRIM(LTRIM(IdSAgencia)), 2)
        ),
        CTE_Stock_All AS (
            SELECT 
                RIGHT('00' + RTRIM(LTRIM(S.IdSAgencia)), 2) AS IdSAgencia, 
                ISNULL(SUM(S.Cartera), 0) AS Cartera, 
                ISNULL(SUM(S.Mora9), 0) AS MoraCPP, 
                ISNULL(SUM(S.Mora31), 0) AS MoraDeficiente, 
                ISNULL(SUM(S.Mora150), 0) AS Mora150, 
                ISNULL(SUM(S.Varios), 0) AS Varios, 
                ISNULL(SUM(CASE WHEN A.Cargo <> 'RECUPERADOR' OR A.Cargo IS NULL THEN S.NroSociosAnterior ELSE 0 END), 0) AS SociosInicio, 
                ISNULL(SUM(CASE WHEN A.Cargo <> 'RECUPERADOR' OR A.Cargo IS NULL THEN S.NroSocios ELSE 0 END), 0) AS SociosActual
            FROM DWH_Gestion_Cartera.dbo.fct_stock S WITH (NOLOCK)
            LEFT JOIN (
                SELECT RTRIM(LTRIM(IdSAsesor)) AS IdSAsesor, MAX(Cargo) AS Cargo
                FROM DWH_Gestion_Cartera.dbo.dim_asesor WITH (NOLOCK)
                WHERE Periodo = @periodo
                GROUP BY RTRIM(LTRIM(IdSAsesor))
            ) A ON RTRIM(LTRIM(S.IdSAsesor)) = A.IdSAsesor
            WHERE S.Periodo = @periodo 
            GROUP BY RIGHT('00' + RTRIM(LTRIM(S.IdSAgencia)), 2)
        ),
        CTE_MetasFlow_All AS (
            SELECT RIGHT('00' + RTRIM(LTRIM(CAST(IdSAgencia AS VARCHAR(5)))), 2) AS IdSAgencia, ISNULL(SUM(CASE WHEN ColocacionNumMetaAjus > 0 THEN ColocacionNumMetaAjus ELSE ColocacionNumMeta END), 0) AS MetaOperaciones 
            FROM dm_productividad.dbo.FctMensual WITH (NOLOCK) WHERE Periodo = @periodo GROUP BY RIGHT('00' + RTRIM(LTRIM(CAST(IdSAgencia AS VARCHAR(5)))), 2)
        ),
        CTE_MetasStock_All AS (
            SELECT 
                RIGHT('00' + RTRIM(LTRIM(CAST(M.IdSAgencia AS VARCHAR(5)))), 2) AS IdSAgencia, 
                ISNULL(MAX(M.Mora9Meta), 0.10) AS MetaMoraCPP, 
                ISNULL(MAX(M.Mora31Meta), CASE WHEN @periodo >= '202610' THEN 0.023 ELSE 0.02 END) AS MetaMoraDeficiente,
                COALESCE(MAX(CI.CarteraInicial), MAX(M.CarteraInicial), 0) AS CarteraInicial
            FROM dm_productividad.dbo.fct_stock_manubl_agencia_month M WITH (NOLOCK)
            LEFT JOIN CTE_CarteraInic_All CI ON RIGHT('00' + RTRIM(LTRIM(CAST(M.IdSAgencia AS VARCHAR(5)))), 2) = CI.IdSAgencia
            WHERE M.Periodo = @periodo 
            GROUP BY RIGHT('00' + RTRIM(LTRIM(CAST(M.IdSAgencia AS VARCHAR(5)))), 2)
        ),
        CTE_TEA_All AS (
            SELECT F.IdSAgencia, CASE WHEN SUM(F.Desembolsos) > 0 THEN SUM(F.Desembolsos * ISNULL(S.MaxTEA, 0)) / SUM(F.Desembolsos) ELSE 0 END AS TEA
            FROM (SELECT RIGHT('00' + RTRIM(LTRIM(IdSAgencia)), 2) AS IdSAgencia, RTRIM(LTRIM(IdSAsesor)) AS IdSAsesor, SUM(ColocacionMontoReal) AS Desembolsos FROM DWH_Gestion_Cartera.dbo.fct_flow WITH (NOLOCK) WHERE Periodo = @periodo GROUP BY RIGHT('00' + RTRIM(LTRIM(IdSAgencia)), 2), RTRIM(LTRIM(IdSAsesor))) F
            LEFT JOIN (SELECT RTRIM(LTRIM(IdSAsesor)) AS IdSAsesor, MAX(TEA) AS MaxTEA FROM DWH_Gestion_Cartera.dbo.fct_stock WITH (NOLOCK) WHERE Periodo = @periodo GROUP BY RTRIM(LTRIM(IdSAsesor))) S ON F.IdSAsesor = S.IdSAsesor
            GROUP BY F.IdSAgencia
        )
        SELECT 
            CASE ISNULL(COALESCE(F.IdSAgencia, S.IdSAgencia, MF.IdSAgencia), '99') WHEN '01' THEN 'Wanchaq' WHEN '02' THEN 'San Jerónimo' WHEN '03' THEN 'Quillabamba' WHEN '04' THEN 'Sicuani' WHEN '05' THEN 'Molino' WHEN '06' THEN 'Juliaca' WHEN '07' THEN 'Lima Los Olivos' WHEN '08' THEN 'Tica Tica' WHEN '09' THEN 'Magisterio' WHEN '10' THEN 'Lima SJL' WHEN '11' THEN 'Chiclayo' WHEN '12' THEN 'Arequipa' WHEN '13' THEN 'Pucallpa' ELSE 'Otra Agencia' END AS agency,
            ((ISNULL(F.Desembolsos, 0) - ISNULL(F.Repagos, 0)) - ISNULL(S.Mora150, 0)) AS crecimientoNeto150, ISNULL(T.TEA, 0) AS tea, ISNULL(MF.MetaOperaciones, 0) AS opTarget, ISNULL(F.NroOper, 0) AS opAchieved,
            CASE WHEN ISNULL(F.NroOper, 0) > 0 THEN (ISNULL(S.Varios, 0) / CAST(F.NroOper AS FLOAT)) / 5.0 ELSE 0 END AS plazo,
            ISNULL(S.SociosInicio, 0) AS sociosInicio, ISNULL(S.SociosActual, 0) AS sociosActual, ISNULL(F.Desembolsos, 0) AS amountAchieved, 
            ISNULL(NULLIF(MS.CarteraInicial, 0), ISNULL(S.Cartera, 0)) AS cartera,
            (ISNULL(MS.MetaMoraCPP, 0.10) * ISNULL(NULLIF(MS.CarteraInicial, 0), ISNULL(S.Cartera, 0))) AS moraCppMax, ISNULL(S.MoraCPP, 0) AS moraCppActual,
            (ISNULL(MS.MetaMoraDeficiente, CASE WHEN @periodo >= '202610' THEN 0.023 ELSE 0.02 END) * ISNULL(NULLIF(MS.CarteraInicial, 0), ISNULL(S.Cartera, 0))) AS moraDefMax, ISNULL(S.MoraDeficiente, 0) AS moraDefActual, ISNULL(F.Repagos, 0) AS repagos
        FROM CTE_Flow_All F FULL OUTER JOIN CTE_Stock_All S ON F.IdSAgencia = S.IdSAgencia FULL OUTER JOIN CTE_MetasFlow_All MF ON COALESCE(F.IdSAgencia, S.IdSAgencia) = MF.IdSAgencia FULL OUTER JOIN CTE_MetasStock_All MS ON COALESCE(F.IdSAgencia, S.IdSAgencia, MF.IdSAgencia) = MS.IdSAgencia FULL OUTER JOIN CTE_TEA_All T ON COALESCE(F.IdSAgencia, S.IdSAgencia, MF.IdSAgencia) = T.IdSAgencia
        WHERE COALESCE(F.IdSAgencia, S.IdSAgencia, MF.IdSAgencia) IS NOT NULL ORDER BY amountAchieved DESC;

        -- =================================================================
        -- 2. AGENCIA COMERCIAL (Solo Analistas Comerciales)
        -- =================================================================
        WITH CTE_Asesores_Com AS (
            SELECT RTRIM(LTRIM(IdSAsesor)) AS IdSAsesor, MAX(RIGHT('00' + RTRIM(LTRIM(IdSAgencia)), 2)) AS IdSAgencia 
            FROM DWH_Gestion_Cartera.dbo.dim_asesor WITH (NOLOCK) 
            WHERE Periodo = @periodo AND (Cargo <> 'RECUPERADOR' OR Cargo IS NULL)
            GROUP BY RTRIM(LTRIM(IdSAsesor))
        ),
        CTE_CarteraInic_Comercial AS (
            SELECT DA.IdSAgencia, SUM(A.CarteraInicial) AS CarteraInicial
            FROM dm_productividad.dbo.fct_stock_manubl_asesor_month A WITH (NOLOCK)
            INNER JOIN CTE_Asesores_Com DA ON RTRIM(LTRIM(A.IdSAsesor)) = DA.IdSAsesor
            WHERE A.Periodo = @periodo
            GROUP BY DA.IdSAgencia
        ),
        CTE_Flow_Com AS (
            SELECT RIGHT('00' + RTRIM(LTRIM(IdSAgencia)), 2) AS IdSAgencia, ISNULL(SUM(ColocacionNumReal), 0) AS NroOper, ISNULL(SUM(ColocacionMontoReal), 0) AS Desembolsos, ISNULL(SUM(RepagoReal), 0) AS Repagos 
            FROM DWH_Gestion_Cartera.dbo.fct_flow WITH (NOLOCK) WHERE Periodo = @periodo AND RTRIM(LTRIM(IdSAsesor)) IN (SELECT IdSAsesor FROM CTE_Asesores_Com) GROUP BY RIGHT('00' + RTRIM(LTRIM(IdSAgencia)), 2)
        ),
        CTE_Stock_Com AS (
            SELECT 
                RIGHT('00' + RTRIM(LTRIM(IdSAgencia)), 2) AS IdSAgencia, 
                ISNULL(SUM(Cartera), 0) AS Cartera, 
                ISNULL(SUM(Mora9), 0) AS MoraCPP, 
                ISNULL(SUM(Mora31), 0) AS MoraDeficiente, 
                ISNULL(SUM(Mora150), 0) AS Mora150, 
                ISNULL(SUM(Varios), 0) AS Varios, 
                ISNULL(SUM(NroSociosAnterior), 0) AS SociosInicio, 
                ISNULL(SUM(NroSocios), 0) AS SociosActual,
                ISNULL(SUM(NroSociosCancelados), 0) AS SociosCancelados,
                ISNULL(SUM(NroSociosRecuperacion), 0) AS NroSociosRecuperacion, 
                ISNULL(SUM(NroSociosActivos), 0) AS SociosActivos
            FROM DWH_Gestion_Cartera.dbo.fct_stock WITH (NOLOCK)
            WHERE Periodo = @periodo AND RTRIM(LTRIM(IdSAsesor)) IN (SELECT IdSAsesor FROM CTE_Asesores_Com) 
            GROUP BY RIGHT('00' + RTRIM(LTRIM(IdSAgencia)), 2)
        ),
        CTE_MetasFlow_Com AS (
            SELECT RIGHT('00' + RTRIM(LTRIM(CAST(IdSAgencia AS VARCHAR(5)))), 2) AS IdSAgencia, ISNULL(SUM(CASE WHEN ColocacionNumMetaAjus > 0 THEN ColocacionNumMetaAjus ELSE ColocacionNumMeta END), 0) AS MetaOperaciones 
            FROM dm_productividad.dbo.FctMensual WITH (NOLOCK) WHERE Periodo = @periodo GROUP BY RIGHT('00' + RTRIM(LTRIM(CAST(IdSAgencia AS VARCHAR(5)))), 2)
        ),
        CTE_MetasStock_Com AS (
            SELECT 
                RIGHT('00' + RTRIM(LTRIM(CAST(M.IdSAgencia AS VARCHAR(5)))), 2) AS IdSAgencia, 
                ISNULL(MAX(M.Mora9Meta), 0.10) AS MetaMoraCPP,

            WHERE M.Periodo = @periodo 
            GROUP BY RIGHT('00' + RTRIM(LTRIM(CAST(M.IdSAgencia AS VARCHAR(5)))), 2)
        ),

            FROM (SELECT RIGHT('00' + RTRIM(LTRIM(IdSAgencia)), 2) AS IdSAgencia, RTRIM(LTRIM(IdSAsesor)) AS IdSAsesor, SUM(ColocacionMontoReal) AS Desembolsos FROM DWH_Gestion_Cartera.dbo.fct_flow WITH (NOLOCK) WHERE Periodo = @periodo AND RTRIM(LTRIM(IdSAsesor)) IN (SELECT IdSAsesor FROM CTE_Asesores_Com) GROUP BY RIGHT('00' + RTRIM(LTRIM(IdSAgencia)), 2), RTRIM(LTRIM(IdSAsesor))) F

            ((ISNULL(F.Desembolsos, 0) - ISNULL(F.Repagos, 0)) - ISNULL(S.Mora150, 0)) AS crecimientoNeto150, ISNULL(T.TEA, 0) AS tea, ISNULL(MF.MetaOperaciones, 0) AS opTarget, ISNULL(F.NroOper, 0) AS opAchieved,
            CASE WHEN ISNULL(F.NroOper, 0) > 0 THEN (ISNULL(S.Varios, 0) / CAST(F.NroOper AS FLOAT)) / 5.0 ELSE 0 END AS plazo,

            ISNULL(NULLIF(MS.CarteraInicial, 0), ISNULL(S.Cartera, 0)) AS cartera,
            (ISNULL(MS.MetaMoraCPP, 0.10) * ISNULL(NULLIF(MS.CarteraInicial, 0), ISNULL(S.Cartera, 0))) AS moraCppMax, ISNULL(S.MoraCPP, 0) AS moraCppActual,
            (ISNULL(MS.MetaMoraDeficiente, CASE WHEN @periodo >= '202610' THEN 0.023 ELSE 0.02 END) * ISNULL(NULLIF(MS.CarteraInicial, 0), ISNULL(S.Cartera, 0))) AS moraDefMax, ISNULL(S.MoraDeficiente, 0) AS moraDefActual,

            ISNULL(S.SociosCancelados, 0) AS sociosCancelados,
            ISNULL(S.NroSociosRecuperacion, 0) AS sociosRecuperacion,
            ISNULL(S.SociosActivos, 0) AS sociosActivos,

            CASE WHEN ISNULL(NULLIF(MS.CarteraInicial, 0), ISNULL(S.Cartera, 0)) > 0 THEN (ISNULL(S.MoraDeficiente, 0) / CAST(ISNULL(NULLIF(MS.CarteraInicial, 0), ISNULL(S.Cartera, 0)) AS FLOAT)) * 100 ELSE 0 END AS pctMoraDef,
            (ISNULL(NULLIF(MS.CarteraInicial, 0), ISNULL(S.Cartera, 0)) + ((ISNULL(F.Desembolsos, 0) - ISNULL(F.Repagos, 0)) - ISNULL(S.Mora150, 0))) AS carteraFin,

        WHERE COALESCE(F.IdSAgencia, S.IdSAgencia, MF.IdSAgencia) IS NOT NULL ORDER BY amountAchieved DESC;

        WITH CTE_Metas_Agencia AS (

            FROM dm_productividad.dbo.fct_stock_manubl_agencia_month WITH (NOLOCK) WHERE Periodo = @periodo GROUP BY RIGHT('00' + RTRIM(LTRIM(CAST(IdSAgencia AS VARCHAR(5)))), 2)

            FROM dm_productividad.dbo.fct_stock_manubl_asesor_month WITH (NOLOCK) WHERE Periodo = @periodo GROUP BY RTRIM(LTRIM(IdSAsesor))
        ),
        CTE_Rec_List AS (
            SELECT RTRIM(LTRIM(IdSAsesor)) AS IdSAsesor, MAX(RIGHT('00' + RTRIM(LTRIM(IdSAgencia)), 2)) AS IdSAgencia, MAX(AsesorNombresApellidos) AS Recuperador 
            FROM DWH_Gestion_Cartera.dbo.dim_asesor WITH (NOLOCK)
            WHERE Periodo = @periodo AND Cargo = 'RECUPERADOR'
            GROUP BY RTRIM(LTRIM(IdSAsesor))
        ),
        CTE_Flow_Rec AS (
            SELECT RTRIM(LTRIM(IdSAsesor)) AS IdSAsesor, ISNULL(SUM(ColocacionMontoReal),0) AS Desembolsos, ISNULL(SUM(RepagoReal), 0) AS Repagos 
            FROM DWH_Gestion_Cartera.dbo.fct_flow WITH (NOLOCK) WHERE Periodo = @periodo GROUP BY RTRIM(LTRIM(IdSAsesor))
        ),
        CTE_Stock_Rec AS (
            SELECT RTRIM(LTRIM(IdSAsesor)) AS IdSAsesor, ISNULL(SUM(Cartera), 0) AS Cartera, ISNULL(SUM(Mora9), 0) AS MoraCPP, ISNULL(SUM(Mora31), 0) AS MoraDeficiente, ISNULL(SUM(Mora150), 0) AS Mora150 
            FROM DWH_Gestion_Cartera.dbo.fct_stock WITH (NOLOCK) WHERE Periodo = @periodo GROUP BY RTRIM(LTRIM(IdSAsesor))
        ),
        CTE_MetasStock_Rec AS (

            FROM CTE_Rec_List R
            LEFT JOIN CTE_Metas_Agencia MA ON R.IdSAgencia = MA.IdSAgencia
            LEFT JOIN CTE_Metas_Asesor MAS ON R.IdSAsesor = MAS.IdSAsesor
        )
        SELECT 
            CASE R.IdSAgencia WHEN '01' THEN 'Wanchaq' WHEN '02' THEN 'San Jerónimo' WHEN '03' THEN 'Quillabamba' WHEN '04' THEN 'Sicuani' WHEN '05' THEN 'Molino' WHEN '06' THEN 'Juliaca' WHEN '07' THEN 'Lima Los Olivos' WHEN '08' THEN 'Tica Tica' WHEN '09' THEN 'Magisterio' WHEN '10' THEN 'Lima SJL' WHEN '11' THEN 'Chiclayo' WHEN '12' THEN 'Arequipa' WHEN '13' THEN 'Pucallpa' ELSE 'Otra Agencia' END AS agency,
            R.Recuperador AS recuperador,
            ((ISNULL(F.Desembolsos, 0) - ISNULL(F.Repagos, 0)) - ISNULL(S.Mora150, 0)) AS crecimientoNeto150,
            ISNULL(NULLIF(MS.CarteraInicial, 0), ISNULL(S.Cartera, 0)) AS carteraInicio,
            (MS.MetaMoraCPP * ISNULL(NULLIF(MS.CarteraInicial, 0), ISNULL(S.Cartera, 0))) AS moraCppMax, ISNULL(S.MoraCPP, 0) AS moraCppActual,
            (MS.MetaMoraDeficiente * ISNULL(NULLIF(MS.CarteraInicial, 0), ISNULL(S.Cartera, 0))) AS moraDefMax, ISNULL(S.MoraDeficiente, 0) AS moraDefActual,
            ISNULL(F.Repagos, 0) AS repagos
        FROM CTE_Rec_List R 
        LEFT JOIN CTE_Flow_Rec F ON R.IdSAsesor = F.IdSAsesor 
        LEFT JOIN CTE_Stock_Rec S ON R.IdSAsesor = S.IdSAsesor 
        LEFT JOIN CTE_MetasStock_Rec MS ON R.IdSAsesor = MS.IdSAsesor
        ORDER BY moraCppActual DESC;
      `);

    const recordsets = result.recordsets as unknown as any[];

    res.json({
      completa: recordsets[0],
      comercial: recordsets[1],
      recuperacion: recordsets[2],
    });
  } catch (error) {
    console.error("Error obteniendo Supervisión:", error);
    res.status(500).json({ error: "Error calculando supervisión" });
  }
});

        FROM [dm_productividad].[dbo].[FctMensual] WITH (NOLOCK)
        WHERE Periodo = @periodo
          AND (@idAgencia = '' OR RIGHT('00' + CAST(IdSAgencia AS VARCHAR(2)), 2) = @idAgencia)
      )
      SELECT
        @periodoAnt AS periodoAnterior,
        ISNULL(ACT.opLogradas, 0) AS opLogradas,
        ISNULL(ANT.opLogradas, 0) AS opLogradasAnt,
        ISNULL(ACT.montoLogrado, 0) AS montoLogrado,
        ISNULL(ANT.montoLogrado, 0) AS montoLogradoAnt,
        ISNULL(ACT.repagos, 0) AS repagos,
        ISNULL(ANT.repagos, 0) AS repagosAnt,
        ISNULL(ACT.opHoy, 0) AS opHoy,
        ISNULL(ACT.montoHoy, 0) AS montoHoy,
        CASE WHEN @asesor <> '' THEN ISNULL(ACT.metaOpIndividual, 30) ELSE MA.metaOpMensual END AS metaOpMensual,
        CASE WHEN @asesor <> '' THEN (ISNULL(ACT.metaOpIndividual, 30) * 2500) ELSE MA.metaMontoMensual END AS metaMontoMensual,
        MA.metaCrecimiento,
        ISNULL(ACT.carteraTotal, 0) AS carteraTotal,
        ISNULL(ANT.carteraTotal, 0) AS carteraTotalAnt,
        (ISNULL(ACT.montoLogrado, 0) - ISNULL(ACT.repagos, 0)) AS crecimientoBruto,
        (ISNULL(ANT.montoLogrado, 0) - ISNULL(ANT.repagos, 0)) AS crecimientoBrutoAnt,
        ((ISNULL(ACT.montoLogrado, 0) - ISNULL(ACT.repagos, 0)) - ISNULL(ACT.mora150, 0)) AS crecimientoNeto150,
        ((ISNULL(ANT.montoLogrado, 0) - ISNULL(ANT.repagos, 0)) - ISNULL(ANT.mora150, 0)) AS crecimientoNeto150Ant,
        ISNULL(ACT.moraCpp, 0) AS moraCpp,
        ISNULL(ANT.moraCpp, 0) AS moraCppAnt,
        ISNULL(ACT.moraDeficiente, 0) AS moraDeficiente,
        ISNULL(ANT.moraDeficiente, 0) AS moraDeficienteAnt,
        ISNULL(ACT.mora150, 0) AS mora150,
        ISNULL(ANT.mora150, 0) AS mora150Ant,
        ISNULL(ACT.sociosInicio, 0) AS sociosInicio,
        ISNULL(ANT.sociosInicio, 0) AS sociosInicioAnt,
        ISNULL(ACT.sociosActuales, 0) AS sociosActuales,
        ISNULL(ANT.sociosActuales, 0) AS sociosActualesAnt,
        (ISNULL(ACT.sociosActuales, 0) - ISNULL(ACT.sociosInicio, 0)) AS sociosNuevosNeto,
        (ISNULL(ANT.sociosActuales, 0) - ISNULL(ANT.sociosInicio, 0)) AS sociosNuevosNetoAnt,
        ISNULL(ACT.sociosActivos, 0) AS sociosActivos,
        ISNULL(ANT.sociosActivos, 0) AS sociosActivosAnt,
        ISNULL(ACT.sociosCancelados, 0) AS sociosCancelados,
        ISNULL(ANT.sociosCancelados, 0) AS sociosCanceladosAnt,
        ISNULL(ACT.sociosTraspasados, 0) AS sociosTraspasados,
        ISNULL(ANT.sociosTraspasados, 0) AS sociosTraspasadosAnt,
        ISNULL(ACT.sociosRecuperacion, 0) AS sociosRecuperacion,
        ISNULL(ANT.sociosRecuperacion, 0) AS sociosRecuperacionAnt,
        (ISNULL(ACT.sociosInicio, 0) - ISNULL(ACT.sociosRecuperacion, 0)) AS sociosInicioMenosRecupe,
        (ISNULL(ANT.sociosInicio, 0) - ISNULL(ANT.sociosRecuperacion, 0)) AS sociosInicioMenosRecupeAnt,
        ISNULL(ACT.teaPromedio, 0) AS teaPromedio,
        ISNULL(ANT.teaPromedio, 0) AS teaPromedioAnt
      FROM MetasAgencia MA
      LEFT JOIN Actual ACT ON 1 = 1
      LEFT JOIN Anterior ANT ON 1 = 1;
    `);

    // 1.3. PRODUCTIVIDAD EN VIVO POR AGENCIA (CÁLCULO AUTOMÁTICO 08:00 AM REDONDEADO A ENTEROS)
    const prodVivoRes = await pool.request().input("periodo", sql.VarChar(6), periodo).query(`
      DECLARE @FechaHoraPeru DATETIME = DATEADD(hour, -5, GETUTCDATE());
      DECLARE @Hoy DATE = CAST(@FechaHoraPeru AS DATE);
      DECLARE @HoraActual TIME = CAST(@FechaHoraPeru AS TIME);

      -- Corte de operaciones logradas al iniciar la jornada laboral (8:00 AM):
      -- Antes de las 8:00 AM toma hasta el día anterior; desde las 8:00 AM calcula el día laboral en curso
      DECLARE @FechaCorteCalculo DATE = CASE
        WHEN @HoraActual >= '08:00:00' THEN @Hoy
        ELSE DATEADD(day, -1, @Hoy)
      END;

      -- Días laborales restantes en el periodo desde el inicio del día laboral (>= 1 para evitar división por 0)
      DECLARE @DiasRestantes INT = (
        SELECT COUNT(Fecha)
        FROM [dm_productividad].[dbo].[dim_calendario] WITH (NOLOCK)
        WHERE Periodo = @periodo
          AND Fecha >= @FechaCorteCalculo
          AND EsDomingo = 0
          AND EsFeriado = 0
      );
      IF ISNULL(@DiasRestantes, 0) < 1 SET @DiasRestantes = 1;

      WITH Agencias AS (
        SELECT * FROM (VALUES
          ('01', 'Wanchaq'), ('02', 'San Jerónimo'), ('03', 'Quillabamba'),
          ('04', 'Sicuani'), ('05', 'Molino'), ('06', 'Juliaca'),
          ('07', 'Lima Los Olivos'), ('08', 'Tica Tica'), ('09', 'Magisterio'),
          ('10', 'Lima SJL'), ('11', 'Chiclayo'), ('12', 'Arequipa'),
          ('13', 'Pucallpa')
        ) AS t(IdSAgencia, NombreAgencia)
      ),
      MetasMensualesAgencia AS (
        SELECT
          RIGHT('00' + RTRIM(LTRIM(CAST(IdSAgencia AS VARCHAR(5)))), 2) AS IdSAgencia,
          ISNULL(SUM(CASE WHEN ISNULL(ColocacionNumMetaAjus, 0) > 0 THEN ColocacionNumMetaAjus ELSE ISNULL(ColocacionNumMeta, 0) END), 0) AS metaOpMes,
          ISNULL(SUM(CASE WHEN ISNULL(CrecimientoMetaAjus, 0) > 0 THEN CrecimientoMetaAjus ELSE ISNULL(CrecimientoMeta, 0) END), 0) AS metaCrecMes,
          ISNULL(SUM(CASE WHEN ISNULL(ColocacionMontoMetaAjus, 0) > 0 THEN ColocacionMontoMetaAjus ELSE ISNULL(ColocacionMontoMeta, 0) END), 0) AS metaMontoMes
        FROM [dm_productividad].[dbo].[FctMensual] WITH (NOLOCK)
        WHERE Periodo = @periodo
        GROUP BY RIGHT('00' + RTRIM(LTRIM(CAST(IdSAgencia AS VARCHAR(5)))), 2)
      ),
      AsesoresCom AS (
        SELECT DISTINCT RTRIM(LTRIM(IdSAsesor)) AS IdSAsesor
        FROM [DWH_Gestion_Cartera].[dbo].[dim_asesor] WITH (NOLOCK)
        WHERE Periodo = @periodo AND (Cargo <> 'RECUPERADOR' OR Cargo IS NULL)
      ),
      LogradoPeriodoInicioDia AS (
        -- Operaciones y desembolsos acumulados en el periodo antes del inicio del día laboral (8:00 AM)
        SELECT
          RIGHT('00' + RTRIM(LTRIM(F.IdSAgencia)), 2) AS IdSAgencia,
          ISNULL(SUM(F.ColocacionNumReal), 0) AS opLogradasMes,
          ISNULL(SUM(F.ColocacionMontoReal), 0) AS montoLogradoMes,
          ISNULL(SUM(F.ColocacionMontoReal - ISNULL(F.RepagoReal, 0)), 0) AS crecBrutoMes
        FROM [DWH_Gestion_Cartera].[dbo].[fct_flow] F WITH (NOLOCK)
        INNER JOIN AsesoresCom AC ON RTRIM(LTRIM(F.IdSAsesor)) = AC.IdSAsesor
        WHERE F.Periodo = @periodo
          AND CAST(F.Fecha AS DATE) < @FechaCorteCalculo
        GROUP BY RIGHT('00' + RTRIM(LTRIM(F.IdSAgencia)), 2)
      ),
      Calculo8AM AS (
        SELECT
          A.IdSAgencia,
          -- Fórmula: ROUND((MetaOpMes - OpLogradas) / DiasRestantes, 0)
          CAST(
            ROUND(
              CASE
                WHEN (ISNULL(MM.metaOpMes, 0) - ISNULL(LP.opLogradasMes, 0)) > 0
                  THEN CAST(ISNULL(MM.metaOpMes, 0) - ISNULL(LP.opLogradasMes, 0) AS FLOAT) / CAST(@DiasRestantes AS FLOAT)
                ELSE 0
              END,
              0
            ) AS INT
          ) AS opFaltantesCalc,
          -- Crecimiento faltante por día redondeado a entero
          CAST(
            ROUND(
              CASE
                WHEN (ISNULL(NULLIF(MM.metaCrecMes, 0), ISNULL(MM.metaMontoMes, 0)) - ISNULL(LP.crecBrutoMes, 0)) > 0
                  THEN CAST(ISNULL(NULLIF(MM.metaCrecMes, 0), ISNULL(MM.metaMontoMes, 0)) - ISNULL(LP.crecBrutoMes, 0) AS FLOAT) / CAST(@DiasRestantes AS FLOAT)
                ELSE 0
              END,
              0
            ) AS INT
          ) AS crecFaltanteCalc
        FROM Agencias A
        LEFT JOIN MetasMensualesAgencia MM ON A.IdSAgencia = MM.IdSAgencia
        LEFT JOIN LogradoPeriodoInicioDia LP ON A.IdSAgencia = LP.IdSAgencia
      ),
      MetasHoy AS (
        SELECT
          RIGHT('00' + CAST(IdSAgencia AS VARCHAR(2)), 2) AS IdSAgencia,
          CAST(ROUND(ISNULL(ColocacionNumMeta, 0), 0) AS INT) AS opFaltantesDia,
          CAST(ROUND(ISNULL(ColocacionMontoMeta, 0), 0) AS INT) AS crecFaltanteDia,
          CAST(ROUND(ISNULL(ColocacionNumProy, 0), 0) AS INT) AS proyOpDia,
          CAST(ROUND(ISNULL(ColocacionMontoProy, 0), 0) AS INT) AS proyMontoDia
        FROM [dm_productividad].[dbo].[FctDiario_MetaProy] WITH (NOLOCK)
        WHERE Fecha = @Hoy
      ),
      ControlHoy AS (
        SELECT
          IdSAgencia,
          CAST(ROUND(Op_Faltantes_Dia, 0) AS INT) AS Op_Faltantes_Dia,
          CAST(ROUND(Crec_Faltante_Dia, 0) AS INT) AS Crec_Faltante_Dia,
          CAST(ROUND(Sugerencia_Op_Gerencia, 0) AS INT) AS Sugerencia_Op_Gerencia,
          CAST(ROUND(Sugerencia_Monto_Gerencia, 0) AS INT) AS Sugerencia_Monto_Gerencia,
          CAST(ROUND(Proy_Op_Admin, 0) AS INT) AS Proy_Op_Admin,
          CAST(ROUND(Proy_Monto_Admin, 0) AS INT) AS Proy_Monto_Admin,
          Actualizado_Por,
          Rol_Editor,
          CONVERT(VARCHAR(19), Fecha_Actualizacion, 120) AS Fecha_Actualizacion
        FROM [DWH_Gestion_Cartera].[dbo].[fct_metas_diarias_control] WITH (NOLOCK)
        WHERE Fecha = @Hoy
      ),
      SumaAsesoresHoy AS (
        SELECT
          IdSAgencia,
          CAST(ROUND(SUM(Meta_Op_Dia), 0) AS INT) AS sumaOpAsesores,
          CAST(ROUND(SUM(Meta_Monto_Dia), 0) AS INT) AS sumaMontoAsesores
        FROM [DWH_Gestion_Cartera].[dbo].[fct_metas_diarias_asesor] WITH (NOLOCK)
        WHERE Fecha = @Hoy
        GROUP BY IdSAgencia
      ),
      LogradoHoy AS (
        SELECT
          RTRIM(LTRIM(IdSAgencia)) AS IdSAgencia,
          CAST(ROUND(ISNULL(SUM(ColocacionNumReal), 0), 0) AS INT) AS colocacionNumReal,
          CAST(ROUND(ISNULL(SUM(ColocacionMontoReal), 0), 0) AS INT) AS colocacionMontoReal
        FROM [DWH_Gestion_Cartera].[dbo].[fct_flow] WITH (NOLOCK)
        WHERE Periodo = @periodo AND CAST(Fecha AS DATE) = @Hoy
        GROUP BY RTRIM(LTRIM(IdSAgencia))
      )
      SELECT
        CONVERT(VARCHAR(10), @Hoy, 120) AS fecha,
        @periodo AS periodo,
        A.IdSAgencia AS idAgencia,
        A.NombreAgencia AS agencia,
        -- Op. Faltantes por Día = cálculo 8 AM (o valor editado manualmente por Gerencia hoy)
        COALESCE(NULLIF(C.Op_Faltantes_Dia, 0), C8.opFaltantesCalc, M.opFaltantesDia, 0) AS opFaltantesDia,
        -- Crec. Faltante por Día redondeado a entero
        COALESCE(NULLIF(C.Crec_Faltante_Dia, 0), C8.crecFaltanteCalc, M.crecFaltanteDia, 0) AS crecFaltanteDia,
        -- Sugerencia Op. Gerencia = Op. Faltantes por Día + 1
        COALESCE(
          NULLIF(C.Sugerencia_Op_Gerencia, 0),
          (COALESCE(NULLIF(C.Op_Faltantes_Dia, 0), C8.opFaltantesCalc, M.opFaltantesDia, 0) + 1)
        ) AS sugerenciaOpGerencia,
        -- Sugerencia Monto Gerencia redondeado a entero
        COALESCE(
          NULLIF(C.Sugerencia_Monto_Gerencia, 0),
          COALESCE(NULLIF(C.Crec_Faltante_Dia, 0), C8.crecFaltanteCalc, M.crecFaltanteDia, 0)
        ) AS sugerenciaMontoGerencia,
        -- Proyección Confirmada por Agencia redondeada a entero
        COALESCE(
          C.Proy_Op_Admin,
          M.proyOpDia,
          C.Sugerencia_Op_Gerencia,
          (COALESCE(NULLIF(C.Op_Faltantes_Dia, 0), C8.opFaltantesCalc, M.opFaltantesDia, 0) + 1)
        ) AS proyeccionColocacion,
        COALESCE(
          C.Proy_Monto_Admin,
          M.proyMontoDia,
          C.Sugerencia_Monto_Gerencia,
          COALESCE(NULLIF(C.Crec_Faltante_Dia, 0), C8.crecFaltanteCalc, M.crecFaltanteDia, 0)
        ) AS proyeccionMonto,
        ISNULL(SA.sumaOpAsesores, 0) AS sumaOpAsesores,
        ISNULL(SA.sumaMontoAsesores, 0) AS sumaMontoAsesores,
        ISNULL(L.colocacionNumReal, 0) AS colocacionNumReal,
        ISNULL(L.colocacionMontoReal, 0) AS colocacionMontoReal,
        ISNULL(C.Actualizado_Por, 'Cálculo Automático 08:00 AM') AS actualizadoPor,
        ISNULL(C.Rol_Editor, '') AS rolEditor,
        ISNULL(C.Fecha_Actualizacion, '') AS fechaActualizacion
      FROM Agencias A
      LEFT JOIN Calculo8AM C8 ON A.IdSAgencia = C8.IdSAgencia
      LEFT JOIN MetasHoy M ON A.IdSAgencia = M.IdSAgencia
      LEFT JOIN ControlHoy C ON A.IdSAgencia = C.IdSAgencia
      LEFT JOIN SumaAsesoresHoy SA ON A.IdSAgencia = SA.IdSAgencia
      LEFT JOIN LogradoHoy L ON A.IdSAgencia = L.IdSAgencia
      ORDER BY A.IdSAgencia ASC;
    `);

    // 1.4. METAS MENSUALES + HISTORIAL
    const metasMensualesRes = await pool.request().input("periodo", sql.VarChar(6), periodo).query(`
      WITH Agencias AS (
        SELECT * FROM (VALUES
          ('01', 'Wanchaq'), ('02', 'San Jerónimo'), ('03', 'Quillabamba'),
          ('04', 'Sicuani'), ('05', 'Molino'), ('06', 'Juliaca'),
          ('07', 'Lima Los Olivos'), ('08', 'Tica Tica'), ('09', 'Magisterio'),
          ('10', 'Lima SJL'), ('11', 'Chiclayo'), ('12', 'Arequipa'),
          ('13', 'Pucallpa')
        ) AS t(IdSAgencia, NombreAgencia)
      )
      SELECT
        @periodo AS periodo,
        A.IdSAgencia AS idAgencia,
        A.NombreAgencia AS agencia,
        ISNULL(M.ColocacionNumMeta, 0) AS colocacionNumMeta,
        ISNULL(M.ColocacionNumMetaAjus, 0) AS colocacionNumMetaAjus,
        ISNULL(M.ColocacionMontoMeta, 0) AS colocacionMontoMeta,
        ISNULL(M.ColocacionMontoMetaAjus, 0) AS colocacionMontoMetaAjus,
        ISNULL(M.NumeroAsesores, 0) AS numeroAsesores,
        ISNULL(M.NumeroAsesoresAjus, 0) AS numeroAsesoresAjus,
        ISNULL(M.CrecimientoMeta, ISNULL(M.CrecimientoMetaAjus, 0)) AS crecimientoMeta,
        ISNULL(M.CrecimientoMetaAjus, 0) AS crecimientoMetaAjus
      FROM Agencias A
      LEFT JOIN [dm_productividad].[dbo].[FctMensual] M WITH (NOLOCK)
        ON RIGHT('00' + CAST(M.IdSAgencia AS VARCHAR(2)), 2) = A.IdSAgencia
        AND M.Periodo = @periodo
      ORDER BY A.IdSAgencia ASC;
    `);

    const historialMetasRes = await pool.request().input("periodo", sql.VarChar(6), periodo).query(`
      SELECT TOP 25
        [Id] AS id,
        [Periodo] AS periodo,
        [Usuario_Nombre] AS usuarioNombre,
        [Usuario_Email] AS usuarioEmail,
        [Usuario_Rol] AS usuarioRol,
        [Resumen_Cambio] AS resumenCambio,
        FORMAT([Fecha_Hora], 'dd/MM/yyyy HH:mm:ss') AS fechaHora
      FROM [DWH_Gestion_Cartera].[dbo].[auditoria_metas_mensuales] WITH (NOLOCK)
      ORDER BY [Fecha_Hora] DESC;
    `);

    // 1.5. TABLA DE ADMINISTRADORES
    const adminsRes = await pool.request().input("periodo", sql.VarChar(6), periodo).query(`
      WITH Agencias AS (
        SELECT * FROM (VALUES
          ('01', 'Wanchaq'), ('02', 'San Jerónimo'), ('03', 'Quillabamba'),
          ('04', 'Sicuani'), ('05', 'Molino'), ('06', 'Juliaca'),
          ('07', 'Lima Los Olivos'), ('08', 'Tica Tica'), ('09', 'Magisterio'),
          ('10', 'Lima SJL'), ('11', 'Chiclayo'), ('12', 'Arequipa'),
          ('13', 'Pucallpa')
        ) AS t(IdSAgencia, NombreAgencia)
      )
      SELECT
        @periodo AS periodo,
        A.IdSAgencia AS idAgencia,
        A.NombreAgencia AS agencia,
        ISNULL(AD.Nombre, '') AS nombre,
        ISNULL(AD.Celular, '') AS celular,
        ISNULL(AD.CorreoPersonal, '') AS correoPersonal,
        ISNULL(AD.CorreoCorporativo, '') AS correoCorporativo
      FROM Agencias A
      LEFT JOIN [DWH_Gestion_Cartera].[dbo].[dim_administrador] AD WITH (NOLOCK)
        ON RIGHT('00' + RTRIM(LTRIM(AD.IdSAgencia)), 2) = A.IdSAgencia
        AND AD.Periodo = @periodo
      ORDER BY A.IdSAgencia ASC;
    `);

    // 1.6. MORA POTENCIAL
    const moraRes = await pool.request()
      .input("idAgencia", sql.VarChar(2), idAgencia)
      .input("asesor", sql.VarChar(150), asesorReq.trim())
      .query(`
        DECLARE @UltimaFecha DATE = (
          SELECT MAX([Fecha_Foto]) FROM [DWH_Gestion_Cartera].[dbo].[fct_riesgo_integral] WITH (NOLOCK)
        );

        SELECT
          R.[IdSAgencia] AS idAgencia,
          CASE R.[IdSAgencia]
            WHEN '01' THEN 'Wanchaq' WHEN '02' THEN 'San Jerónimo' WHEN '03' THEN 'Quillabamba'
            WHEN '04' THEN 'Sicuani' WHEN '05' THEN 'Molino' WHEN '06' THEN 'Juliaca'
            WHEN '07' THEN 'Lima Los Olivos' WHEN '08' THEN 'Tica Tica' WHEN '09' THEN 'Magisterio'
            WHEN '10' THEN 'Lima SJL' WHEN '11' THEN 'Chiclayo' WHEN '12' THEN 'Arequipa'
            WHEN '13' THEN 'Pucallpa' ELSE 'Otra Agencia'
          END AS agencia,
          RTRIM(LTRIM(R.[IdSAsesor])) AS idAsesor,
          ISNULL(D_AS.NombreAsesor, RTRIM(LTRIM(R.[IdSAsesor]))) AS asesor,
          RTRIM(LTRIM(R.[Pagare])) AS pagare,
          CASE
            WHEN R.[Cuenta] IS NULL OR LTRIM(RTRIM(R.[Cuenta])) = '' THEN ''
            ELSE RIGHT('000000000000' + LTRIM(RTRIM(R.[Cuenta])), 12)
          END AS cuenta,
          RTRIM(LTRIM(R.[Socio])) AS socio,
          ISNULL(DirSoc.Direccion_Socio, '') AS direccion,
          RTRIM(LTRIM(ISNULL(R.[Producto], ''))) AS producto,
          ISNULL(R.[Saldo_Capital], 0) AS saldoCapital,
          ISNULL(R.[Cuota_Mensual], 0) AS cuotaMensual,
          ISNULL(R.[Cuotas_Atrasadas], 0) AS cuotasAtrasadas,
          ISNULL(R.[Dias_Atraso], 0) AS diasAtraso,
          ISNULL(R.[Cuotas_Pendientes], 0) AS cuotasPendientes,
          ISNULL(R.[Cuotas_Pagadas], 0) AS cuotasPagadas,
          ISNULL(R.[Avance_Cuota], '0%') AS avanceCuota,
          ISNULL(R.[Telefonos], '') AS telefonos,
          R.[Probabilidad_IA] AS probabilidadIa,
          COALESCE(CW.[Compromiso_Actual], R.[Compromiso], '') AS compromiso,
          CASE
            WHEN CW.[Fecha_Promesa_Actual] IS NOT NULL THEN CONVERT(VARCHAR(10), CW.[Fecha_Promesa_Actual], 120)
            WHEN R.[Fecha_Promesa] IS NOT NULL THEN CONVERT(VARCHAR(10), R.[Fecha_Promesa], 120)
            ELSE ''
          END AS fechaPromesa,
          ISNULL(CW.[Autor_Actual], CASE WHEN R.[Compromiso] IS NOT NULL AND R.[Compromiso] <> '' THEN 'Sincronizado GSheets' ELSE '' END) AS autorActual,
          ISNULL(CW.[Rol_Actual], '') AS rolActual,
          CASE WHEN CW.[Fecha_Actualizacion] IS NOT NULL THEN FORMAT(CW.[Fecha_Actualizacion], 'dd/MM/yyyy HH:mm') ELSE '' END AS fechaActualizacionCompromiso,
          ISNULL(CW.[Compromiso_Anterior], '') AS compromisoAnterior,
          CASE WHEN CW.[Fecha_Promesa_Anterior] IS NOT NULL THEN CONVERT(VARCHAR(10), CW.[Fecha_Promesa_Anterior], 103) ELSE '' END AS fechaPromesaAnterior,
          ISNULL(CW.[Autor_Anterior], '') AS autorAnterior,
          ISNULL(CW.[Rol_Anterior], '') AS rolAnterior,
          CASE WHEN CW.[Fecha_Anterior] IS NOT NULL THEN FORMAT(CW.[Fecha_Anterior], 'dd/MM/yyyy HH:mm') ELSE '' END AS fechaAnteriorCompromiso
        FROM [DWH_Gestion_Cartera].[dbo].[fct_riesgo_integral] R WITH (NOLOCK)
        LEFT JOIN [DWH_Gestion_Cartera].[dbo].[fct_compromisos_mora_web] CW WITH (NOLOCK)
          ON RTRIM(LTRIM(R.[Pagare])) = RTRIM(LTRIM(CW.[Pagare]))
        OUTER APPLY (

          ORDER BY DA.[Periodo] DESC
        ) D_AS
        OUTER APPLY (
          SELECT TOP 1 LTRIM(RTRIM(ISNULL(S.[DIRECCION], ''))) AS Direccion_Socio
          FROM [TRANSACMIF].[dbo].[SOCIOS] S WITH (NOLOCK)
          WHERE S.[COD_SOCIO] = RIGHT('000000000000' + LTRIM(RTRIM(R.[Cuenta])), 12)
             OR S.[RAZON] = RTRIM(LTRIM(R.[Socio]))
        ) DirSoc
        WHERE R.[Fecha_Foto] = @UltimaFecha
          AND R.[Estado_Mora] = '3. Mora Potencial'
          AND (@idAgencia = '' OR RTRIM(LTRIM(R.[IdSAgencia])) = @idAgencia)
          AND (
            @asesor = ''
            OR RTRIM(LTRIM(R.[IdSAsesor])) = @asesor
            OR RTRIM(LTRIM(D_AS.NombreAsesor)) LIKE '%' + @asesor + '%'
          )
        ORDER BY R.[Saldo_Capital] DESC;
      `);

    // 1.7. META DIARIA PROPIA DEL ANALISTA
    const metaAsesorHoyRes = await pool.request()
      .input("asesor", sql.VarChar(150), asesorReq.trim())
      .query(`
        SELECT TOP 1
          [Meta_Op_Dia] AS metaOpDia,
          [Meta_Monto_Dia] AS metaMontoDia,
          FORMAT([Fecha_Actualizacion], 'dd/MM/yyyy HH:mm') AS fechaActualizacion
        FROM [DWH_Gestion_Cartera].[dbo].[fct_metas_diarias_asesor] WITH (NOLOCK)
        WHERE [Fecha] = CAST(DATEADD(hour, -5, GETUTCDATE()) AS DATE)
          AND ([IdSAsesor] = @asesor OR [AsesorNombre] LIKE '%' + @asesor + '%')
      `);

    // 1.8. Hora oficial de última actualización en el DWH
    const logRes = await pool.request().query(`
      SELECT TOP (1)
        FORMAT([UltimaActualizacion], 'dd/MM/yyyy · hh:mm:ss tt', 'es-PE') AS UltimaActualizacion
      FROM [DWH_Gestion_Cartera].[dbo].[Log_Actualizacion] WITH (NOLOCK)
      ORDER BY [UltimaActualizacion] DESC
    `);
    const ultimaActualizacionDwh = logRes.recordset?.[0]?.UltimaActualizacion || "";

    res.json({
      periodosDisponibles,
      periodoActivo: periodo,
      ultimaActualizacionDwh,
      versionSignature: getCombinedVersionSignature(),
      kpis: kpiResult.recordset?.[0] || {},
      productividadDiaria: prodVivoRes.recordset || [],
      metasMensuales: metasMensualesRes.recordset || [],
      historialMetasMensuales: historialMetasRes.recordset || [],
      administradores: adminsRes.recordset || [],
      moraPotencial: moraRes.recordset || [],
      metaAsesorHoy: metaAsesorHoyRes.recordset?.[0] || { metaOpDia: 0, metaMontoDia: 0, fechaActualizacion: "" },
    });
  } catch (error) {
    console.error("Error en /api/welcome/init:", error);
    res.status(500).json({ error: "Error cargando el centro de mando Welcome" });

router.post(["/welcome/metas-mensuales", "/api/welcome/metas-mensuales"], async (req: Request, res: Response) => {
  try {
    await ensureWelcomeTables();
    const { periodo, filas, usuarioNombre, usuarioEmail, usuarioRol } = req.body;
    if (!periodo || !Array.isArray(filas)) {
      return res.status(400).json({ error: "Periodo y filas son obligatorios" });
    }

    // Asegurar que exista la columna CrecimientoMeta en FctMensual para conservar el histórico de variación
    await pool.request().query(`
      IF COL_LENGTH('[dm_productividad].[dbo].[FctMensual]', 'CrecimientoMeta') IS NULL
        ALTER TABLE [dm_productividad].[dbo].[FctMensual] ADD [CrecimientoMeta] FLOAT NULL;
    `);

    const transaction = new sql.Transaction(pool);
    await transaction.begin();

    try {
      await new sql.Request(transaction)
        .input("periodo", sql.VarChar(6), String(periodo).trim())
        .query(`DELETE FROM [dm_productividad].[dbo].[FctMensual] WHERE [Periodo] = @periodo`);

      let totalOpBase = 0;
      let totalOpAjus = 0;
      let totalMontoBase = 0;
      let totalMontoAjus = 0;

      for (const row of filas) {
        const opMeta = Number(row.colocacionNumMeta || 0);
        const opAjus = Number(row.colocacionNumMetaAjus ?? opMeta);
        const montoMeta = Number(row.colocacionMontoMeta || 0);
        const montoAjus = Number(row.colocacionMontoMetaAjus ?? montoMeta);
        const asesMeta = Number(row.numeroAsesores || 0);
        const asesAjus = Number(row.numeroAsesoresAjus ?? asesMeta);
        const crecMeta = Number(row.crecimientoMeta ?? row.crecimientoMetaAjus ?? 0);
        const crecAjus = Number(row.crecimientoMetaAjus ?? crecMeta);

        totalOpBase += opMeta;
        totalOpAjus += opAjus;
        totalMontoBase += montoMeta;
        totalMontoAjus += montoAjus;

        await new sql.Request(transaction)
          .input("periodo", sql.VarChar(6), String(periodo).trim())
          .input("idAge", sql.Int, parseInt(String(row.idAgencia || "1"), 10))
          .input("opMeta", sql.Int, opMeta)
          .input("opAjus", sql.Int, opAjus)
          .input("montoMeta", sql.Float, montoMeta)
          .input("montoAjus", sql.Float, montoAjus)
          .input("asesores", sql.Int, asesMeta)
          .input("asesoresAjus", sql.Int, asesAjus)
          .input("crecMeta", sql.Float, crecMeta)
          .input("crecAjus", sql.Float, crecAjus)
          .query(`
            INSERT INTO [dm_productividad].[dbo].[FctMensual]
              ([Periodo], [IdSAgencia], [ColocacionNumMeta], [ColocacionNumMetaAjus], [ColocacionMontoMeta], [ColocacionMontoMetaAjus], [NumeroAsesores], [NumeroAsesoresAjus], [CrecimientoMeta], [CrecimientoMetaAjus])
            VALUES
              (@periodo, @idAge, @opMeta, @opAjus, @montoMeta, @montoAjus, @asesores, @asesoresAjus, @crecMeta, @crecAjus)
          `);
      }

      const diffOp = totalOpAjus - totalOpBase;
      const diffSign = diffOp >= 0 ? `+${diffOp}` : `${diffOp}`;
      const resumen = `Actualizó metas del periodo ${periodo} (${filas.length} agencias · Meta Op Base: ${totalOpBase} | Ajustada: ${totalOpAjus} [${diffSign} op.] · Monto Ajustado: S/ ${totalMontoAjus.toLocaleString("es-PE")})`;

      await new sql.Request(transaction)
        .input("periodo", sql.VarChar(6), String(periodo).trim())
        .input("nombre", sql.VarChar(150), String(usuarioNombre || "Super Admin"))
        .input("email", sql.VarChar(150), String(usuarioEmail || ""))
        .input("rol", sql.VarChar(60), String(usuarioRol || "SUPER_ADMIN"))
        .input("resumen", sql.NVarChar(sql.MAX), resumen)
        .query(`
          INSERT INTO [DWH_Gestion_Cartera].[dbo].[auditoria_metas_mensuales]
            ([Periodo], [Usuario_Nombre], [Usuario_Email], [Usuario_Rol], [Resumen_Cambio], [Fecha_Hora])
          VALUES (@periodo, @nombre, @email, @rol, @resumen, GETDATE());

          IF EXISTS (SELECT 1 FROM [DWH_Gestion_Cartera].[dbo].[Log_Actualizacion] WHERE [Id] = 1)
            UPDATE [DWH_Gestion_Cartera].[dbo].[Log_Actualizacion] SET [UltimaActualizacion] = GETDATE() WHERE [Id] = 1;
        `);

      await transaction.commit();
      res.json({ ok: true, message: "Metas mensuales guardadas y registradas en historial." });
    } catch (err) {
      await transaction.rollback();
      throw err;
    }
  } catch (error) {

    res.status(500).json({ error: "Error guardando metas mensuales" });

router.get(["/welcome/config-sistema", "/api/welcome/config-sistema"], async (_req: Request, res: Response) => {
  try {
    await ensureConfigSistemaTable();
    const result = await pool.request().query(`
      SELECT TOP 1
        [Config_Json],
        ISNULL([Actualizado_Por], 'Valores por Defecto') AS Actualizado_Por,
        ISNULL([Usuario_Rol], 'SISTEMA') AS Usuario_Rol,
        FORMAT([Fecha_Actualizacion], 'dd/MM/yyyy HH:mm:ss') AS Fecha_Actualizacion
      FROM [DWH_Gestion_Cartera].[dbo].[cfg_sistema_control] WITH (NOLOCK)
      WHERE [Id] = 1;
    `);

    if (!result.recordset.length) {
      return res.json({
        config: DEFAULT_SYSTEM_CONFIG,
        actualizadoPor: "Sistema (Por defecto)",
        fechaActualizacion: "",
      });
    }

    const row = result.recordset[0];
    let parsed = DEFAULT_SYSTEM_CONFIG;
    try {
      const dbJson = JSON.parse(row.Config_Json);
      parsed = {
        daemon: { ...DEFAULT_SYSTEM_CONFIG.daemon, ...(dbJson.daemon || {}) },
        polling: { ...DEFAULT_SYSTEM_CONFIG.polling, ...(dbJson.polling || {}) },
        permisos: { ...DEFAULT_SYSTEM_CONFIG.permisos, ...(dbJson.permisos || {}) },
      };
    } catch {}

    res.json({
      config: parsed,
      actualizadoPor: row.Actualizado_Por,
      usuarioRol: row.Usuario_Rol,
      fechaActualizacion: row.Fecha_Actualizacion,
    });
  } catch (error) {
    console.error("Error obteniendo config-sistema:", error);
    res.status(500).json({ error: "Error leyendo configuración del sistema" });
  }
});

router.post(["/welcome/config-sistema", "/api/welcome/config-sistema"], async (req: Request, res: Response) => {
  try {
    await ensureConfigSistemaTable();
    const { config, usuarioNombre, usuarioRol } = req.body;
    if (!config) {
      return res.status(400).json({ error: "El objeto config es requerido" });
    }

    const jsonStr = JSON.stringify(config);
    await pool.request()
      .input("json", sql.NVarChar(sql.MAX), jsonStr)
      .input("autor", sql.VarChar(150), String(usuarioNombre || "Super Admin"))
      .input("rol", sql.VarChar(60), String(usuarioRol || "SUPER_ADMIN"))
      .query(`
        IF EXISTS (SELECT 1 FROM [DWH_Gestion_Cartera].[dbo].[cfg_sistema_control] WHERE [Id] = 1)
          UPDATE [DWH_Gestion_Cartera].[dbo].[cfg_sistema_control]
          SET [Config_Json] = @json,
              [Actualizado_Por] = @autor,
              [Usuario_Rol] = @rol,
              [Fecha_Actualizacion] = GETDATE()
          WHERE [Id] = 1;
        ELSE
          INSERT INTO [DWH_Gestion_Cartera].[dbo].[cfg_sistema_control]
            ([Id], [Config_Json], [Actualizado_Por], [Usuario_Rol], [Fecha_Actualizacion])
          VALUES (1, @json, @autor, @rol, GETDATE());
      `);

    res.json({ ok: true, message: "Configuración de horarios y accesos guardada en SQL Server." });
  } catch (error) {
    console.error("Error guardando config-sistema:", error);
    res.status(500).json({ error: "No se pudo guardar la configuración del sistema" });
  }
});
